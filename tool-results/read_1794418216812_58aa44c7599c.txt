    99→async function api<T = Record<string, unknown>>(path: string, options: RequestInit = {}): Promise<T> {
   100→  const headers = options.body instanceof FormData ? {} : { 'Content-Type': 'application/json' };
   101→  const res = await fetch(path, { ...options, headers: { ...headers, ...(options.headers as Record<string, string> | undefined) } });
   102→  if (!res.ok) {
   103→    let detail = `HTTP ${res.status}`;
   104→    try {
   105→      const err = await res.json();
   106→      detail = (err as { detail?: string }).detail || detail;
   107→    } catch { /* ignore */ }
   108→    throw new Error(detail);
   109→  }
   110→  return (await res.json()) as T;
   111→}

   189→  async bootstrap() {
   190→    this.emit();
   191→    // Yield one tick so React StrictMode's double-mount cleanup lands before
   192→    // the heavy 14 MB character download starts (prevents a duplicate load).
   193→    await new Promise((r) => setTimeout(r, 0));
   194→    const sanctuary = this.sanctuary;
   195→    if (!sanctuary || this.disposed) return;
   196→
   197→    try {
   198→      await sanctuary.initCharacter((xhr: { loaded: number; total: number } | undefined) => {
   199→        if (this.disposed) return;
   200→        if (xhr && xhr.total) {
   201→          const pct = Math.round((xhr.loaded / xhr.total) * 100);
   202→          this.loadingText = `Loading rigged 3D character & skeletal clips (${pct}%)…`;
   203→          this.emit();
   204→        }
   205→      });
   206→    } catch (err) {
   207→      this.loadingText = `Character asset warning: ${(err as Error).message}`;
   208→      this.emit();
   209→    }
   210→    if (this.disposed) return;
   211→
   212→    try {
   213→      const [worldSnap, scenariosRes] = await Promise.all([
   214→        api<WorldSnapshot>('/api/v1/world'),
   215→        api<{ scenarios: ScenarioDef[] }>('/api/v1/scenarios'),
   216→      ]);
   217→      if (this.disposed) return;
   218→      this.scenarios = scenariosRes.scenarios || [];
   219→      this.applyWorldState(worldSnap, true);
   220→    } catch (err) {
   221→      this.error = (err as Error).message;
   222→      this.emit();
   223→      return;
   224→    }
   225→
   226→    this.ready = true;
   227→    this.loadingText = 'NEXUS online.';
   228→    this.emit();
   229→
   230→    // Animation loop
   231→    sanctuary.renderer.setAnimationLoop(() => {
   232→      const dt = this.animClock.getDelta();
   233→      sanctuary.renderFrame(dt, this.opticalCamEl, Boolean(this.world?.switches.camera));
   234→      this.tickScenarioUI();
   235→    });
   236→
   237→    // Telemetry poll
   238→    this.telemetryTimer = setInterval(async () => {
   239→      try {
   240→        this.telemetry = await api<Telemetry>('/api/v1/telemetry');
   241→        sanctuary.updateTelemetryBoard({
   242→          ...this.telemetry,
   243→          inference: this.telemetry.inference,
   244→        } as never);
   245→        this.emit();
   246→      } catch { /* transient */ }
   247→    }, 8000);
   248→    void this.fetchTelemetryOnce();
   249→  }
   250→
   251→  private async fetchTelemetryOnce() {
   252→    try {
   253→      this.telemetry = await api<Telemetry>('/api/v1/telemetry');
   254→      this.sanctuary?.updateTelemetryBoard(this.telemetry as never);
   255→      this.emit();
   256→    } catch { /* ignore */ }
   257→  }
   258→
   259→  dispose() {
   260→    this.disposed = true;
   261→    if (this.telemetryTimer) clearInterval(this.telemetryTimer);
   262→    this.sanctuary?.renderer.setAnimationLoop(null);
   263→  }

   265→  // ------------------------------------------------------------------
   266→  // World state application
   267→  // ------------------------------------------------------------------
   268→  applyWorldState(snap: WorldSnapshot, initialSnap = false) {
   269→    if (!snap || !this.sanctuary) return;
   270→    this.world = snap;
   271→    const sanctuary = this.sanctuary;
   272→    sanctuary.reducedMotion = Boolean(snap.reduced_motion);
   273→
   274→    const zoneInfo = snap.zones?.[snap.current_zone];
   275→    if (zoneInfo?.look_target) {
   276→      sanctuary.jarvis.lookWorldTarget.set(...(zoneInfo.look_target as [number, number, number]));
   277→    }
   278→    sanctuary.jarvis.focusCamera = ['conversational_focus', 'wake_auth', 'digital_human_embodiment', 'holographic_presence'].includes(snap.active_mode);
   279→
   280→    const postureMap: Record<string, string> = {
   281→      simulation_stage: snap.active_scenario ? 'scenario_direct' : 'scenario_construct',
   282→      media_presentation: 'video_present',
   283→      central_operations: snap.active_mode === 'conversational_focus' ? 'conversational_focus' : 'console_interaction',
   284→    };
   285→    const posture = postureMap[snap.current_zone] || 'console_interaction';
   286→    const pos = snap.position;
   287→
   288→    if (initialSnap) {
   289→      sanctuary.jarvis.root.position.set(pos.x, 0, pos.z);
   290→      sanctuary.jarvis.root.rotation.y = pos.yaw;
   291→      sanctuary.jarvis.targetYaw = pos.yaw;
   292→      sanctuary.jarvis.postureMode = posture;
   293→      sanctuary.jarvis.emotionalState = snap.emotional_state;
   294→    } else {
   295→      sanctuary.jarvis.navigateTo(pos.x, pos.z, pos.yaw, {
   296→        urgent: snap.locomotion_state === 'running' || snap.emotional_state === 'urgent',
   297→        posture,
   298→        emotion: snap.emotional_state || 'focused',
   299→      });
   300→    }
   301→
   302→    const isHolo = snap.active_mode === 'holographic_presence';
   303→    sanctuary.setHologramMode(isHolo, this.holoLayout);
   304→
   305→    if (snap.active_camera_preset) {
   306→      sanctuary.cameraDirector.setPreset(snap.active_camera_preset, sanctuary.reducedMotion);
   307→    }
   308→
   309→    if (snap.active_scenario && !sanctuary.scenarioDirector.activeScenario) {
   310→      const def = this.scenarios.find((s) => s.id === snap.active_scenario.scenario_id);
   311→      if (def) this.activateClientScenario(def, snap.active_scenario);
   312→    } else if (!snap.active_scenario && sanctuary.scenarioDirector.activeScenario) {
   313→      sanctuary.scenarioDirector.clearStage();
   314→      this.activeScenarioDef = null;
   315→    }
   316→
   317→    sanctuary.syncPhysicalSwitchLevers(snap.switches || {});
   318→    void this.enforceBrowserSensorGates();
   319→    this.emit();
   320→  }

   325→  async sendConversation(content: string, modality: 'text' | 'voice' | 'vision' = 'text') {
   326→    try {
   327→      const payload = await api<{
   328→        jarvis_message: JarvisMessage;
   329→        world: WorldSnapshot;
   330→        activated_scenario: ScenarioRuntime | null;
   331→      }>('/api/v1/conversations/send', {
   332→        method: 'POST',
   333→        body: JSON.stringify({ content, modality }),
   334→      });
   335→
   336→      this.applyWorldState(payload.world, false);
   337→      this.showJarvisResponse(payload.jarvis_message);
   338→
   339→      if (payload.activated_scenario) {
   340→        const def = this.scenarios.find((s) => s.id === payload.activated_scenario.scenario_id);
   341→        if (def) this.activateClientScenario(def, payload.activated_scenario);
   342→      }
   343→    } catch (err) {
   344→      this.error = (err as Error).message;
   345→      this.emit();
   346→      setTimeout(() => { this.error = null; this.emit(); }, 4200);
   347→    }
   348→  }

   350→  showJarvisResponse(jarvisMsg: JarvisMessage | null) {
   351→    if (!jarvisMsg || !this.sanctuary) return;
   352→    this.subtitle = jarvisMsg;
   353→    this.interruptAvailable = true;
   354→    this.speakingState = true;
   355→    this.emit();
   356→
   357→    if (this.world?.switches.embodiment !== false) {
   358→      this.sanctuary.jarvis.speak(jarvisMsg.content, {
   359→        audio: jarvisMsg.audio ? { data: jarvisMsg.audio, mime: jarvisMsg.audio_mime || 'audio/wav' } : undefined,
   360→        onEnd: () => {
   361→          this.interruptAvailable = false;
   362→          this.speakingState = false;
   363→          this.emit();
   364→        },
   365→      });
   366→    }
   367→  }

   522→  toggleScenarioRecording(): boolean {
   523→    if (this.mediaRecorder && this.mediaRecorder.state === 'recording') {
   524→      this.mediaRecorder.stop();
   525→      this.recording = false;
   526→      this.emit();
   527→      return false;
   528→    }
   529→    if (!this.canvas.captureStream || typeof MediaRecorder === 'undefined') {
   530→      this.error = 'MediaRecorder WebM capture is not supported by this browser.';
   531→      this.emit();
   532→      setTimeout(() => { this.error = null; this.emit(); }, 3600);
   533→      return false;
   534→    }
   535→    const stream = this.canvas.captureStream(30);
   536→    const recorder = new MediaRecorder(stream, { mimeType: 'video/webm' });
   537→    this.recordedChunks = [];
   538→    recorder.ondataavailable = (ev) => {
   539→      if (ev.data.size > 0) this.recordedChunks.push(ev.data);
   540→    };
   541→    recorder.onstop = () => {
   542→      const blob = new Blob(this.recordedChunks, { type: 'video/webm' });
   543→      const url = URL.createObjectURL(blob);
   544→      const a = document.createElement('a');
   545→      a.href = url;
   546→      a.download = 'jarvis_nexus_scenario.webm';
   547→      a.click();
   548→    };
   549→    recorder.start();
   550→    this.mediaRecorder = recorder;
   551→    this.recording = true;
   552→    this.emit();
   553→    return true;
   554→  }

   556→  private tickScenarioUI() {
   557→    const sd = this.sanctuary?.scenarioDirector;
   558→    if (!sd || !sd.activeScenario) {
   559→      if (this.scenarioTimeReadout) {
   560→        this.scenarioTimeReadout = '';
   561→        this.scenarioPhase = '';
   562→        this.emit();
   563→      }
   564→      return;
   565→    }
   566→    const now = performance.now();
   567→    if (now - this.lastTickEmit < 200) return;
   568→    this.lastTickEmit = now;
   569→
   570→    this.scenarioTimeReadout = `${sd.currentTime.toFixed(1).padStart(4, '0')}s / ${sd.duration.toFixed(1)}s`;
   571→    const tl = sd.activeScenario.timeline || [];
   572→    let activeCue = tl[0];
   573→    for (const cue of tl) {
   574→      if (sd.currentTime >= cue.t) activeCue = cue;
   575→    }
   576→    if (activeCue) this.scenarioPhase = activeCue.phase;
   577→    this.emit();
   578→  }

   580→  get scenarioDirector() {
   581→    return this.sanctuary?.scenarioDirector ?? null;
   582→  }

   584→  // ------------------------------------------------------------------
   585→  // Real browser sensor enforcement (no hidden capture)
   586→  // ------------------------------------------------------------------
   587→  async enforceBrowserSensorGates() {
   588→    if (!this.sanctuary || !this.world) return;
   589→    const sw = this.world.switches;
   590→
   591→    if (!sw.microphone || !sw.master) {
   592→      if (this.micStream) {
   593→        this.micStream.getTracks().forEach((t) => t.stop());
   594→        this.micStream = null;
   595→      }
   596→    }
   597→
   598→    if (!sw.camera || !sw.master) {
   599→      if (this.camStream) {
   600→        this.camStream.getTracks().forEach((t) => t.stop());
   601→        this.camStream = null;
   602→        if (this.opticalCamEl) this.opticalCamEl.srcObject = null;
   603→        this.camState = 'off';
   604→      }
   605→      this.sanctuary.updateVisionBoard(false, null);
   606→    } else if (!this.camStream && navigator.mediaDevices?.getUserMedia) {
   607→      try {
   608→        this.camStream = await navigator.mediaDevices.getUserMedia({ video: { width: 640, height: 360 } });
   609→        if (this.opticalCamEl) {
   610→          this.opticalCamEl.srcObject = this.camStream;
   611→          await this.opticalCamEl.play().catch(() => undefined);
   612→        }
   613→        this.sanctuary.updateVisionBoard(true, this.opticalCamEl);
   614→        this.camState = 'live';
   615→      } catch {
   616→        this.sanctuary.updateVisionBoard(false, null);
   617→        this.camState = 'off';
   618→      }
   619→    }
   620→    this.emit();
   621→  }

   623→  // ------------------------------------------------------------------
   624→  // Voice input: SpeechRecognition preferred, MediaRecorder+ASR fallback
   625→  // ------------------------------------------------------------------
   626→  async startVoiceCapture(): Promise<void> {
   627→    if (!this.world?.switches.microphone) {
   628→      this.error = 'Microphone Switch (#2) is OFF. Enable MIC in the top switch bar to authorise audio capture.';
   629→      this.emit();
   630→      setTimeout(() => { this.error = null; this.emit(); }, 4200);
   631→      return;
   632→    }
   633→
   634→    const SR = (window as unknown as { SpeechRecognition?: new () => SpeechRecognition; webkitSpeechRecognition?: new () => SpeechRecognition }).SpeechRecognition
   635→      ?? (window as unknown as { webkitSpeechRecognition?: new () => SpeechRecognition }).webkitSpeechRecognition;
   636→
   637→    if (SR) {
   638→      const rec = new SR();
   639→      rec.lang = 'en-US';
   640→      rec.interimResults = false;
   641→      this.micState = 'listening';
   642→      this.emit();
   643→      rec.onresult = async (ev: SpeechRecognitionEvent) => {
   644→        const transcript = ev.results[0][0].transcript;
   645→        this.micState = 'processing';
   646→        this.emit();
   647→        await this.sendConversation(transcript, 'voice');
   648→        this.micState = 'off';
   649→        this.emit();
   650→      };
   651→      rec.onerror = () => {
   652→        this.micState = 'off';
   653→        this.emit();
   654→      };
   655→      rec.onend = () => {
   656→        if (this.micState === 'listening') {
   657→          this.micState = 'off';
   658→          this.emit();
   659→        }
   660→      };
   661→      try { rec.start(); } catch { /* already started */ }
   662→      return;
   663→    }
   664→
   665→    // Fallback: record webm/opus, transcribe via CORE ASR
   666→    if (!navigator.mediaDevices?.getUserMedia || typeof MediaRecorder === 'undefined') {
   667→      this.error = 'Voice capture is unavailable in this browser. Type directly instead.';
   668→      this.emit();
   669→      setTimeout(() => { this.error = null; this.emit(); }, 4200);
   670→      return;
   671→    }
   672→    try {
   673→      this.micStream = await navigator.mediaDevices.getUserMedia({ audio: true });
   674→      const recorder = new MediaRecorder(this.micStream);
   675→      const chunks: Blob[] = [];
   676→      this.micState = 'listening';
   677→      this.emit();
   678→      recorder.ondataavailable = (ev) => { if (ev.data.size > 0) chunks.push(ev.data); };
   679→      recorder.onstop = async () => {
   680→        this.micStream?.getTracks().forEach((t) => t.stop());
   681→        this.micState = 'processing';
   682→        this.emit();
   683→        try {
   684→          const blob = new Blob(chunks, { type: chunks[0]?.type || 'audio/webm' });
   685→          const b64 = await blobToBase64(blob);
   686→          const res = await api<{ text: string }>('/api/v1/asr', {
   687→            method: 'POST',
   688→            body: JSON.stringify({ audio_base64: b64 }),
   689→          });
   690→          if (res.text) {
   691→            await this.sendConversation(res.text, 'voice');
   692→          } else {
   693→            this.error = 'No speech detected.';
   694→            this.emit();
   695→            setTimeout(() => { this.error = null; this.emit(); }, 3000);
   696→          }
   697→        } catch (err) {
   698→          this.error = `Voice transcription failed: ${(err as Error).message}`;
   699→          this.emit();
   700→          setTimeout(() => { this.error = null; this.emit(); }, 4200);
   701→        }
   702→        this.micState = 'off';
   703→        this.emit();
   704→      };
   705→      recorder.start();
   706→      // Auto-stop after 12s safety window
   707→      setTimeout(() => {
   708→        if (recorder.state === 'recording') recorder.stop();
   709→      }, 12000);
   710→      this._pendingRecorder = recorder;
   711→    } catch {
   712→      this.micState = 'off';
   713→      this.error = 'Microphone permission denied.';
   714→      this.emit();
   715→      setTimeout(() => { this.error = null; this.emit(); }, 3600);
   716→    }
   717→  }
   718→
   719→  private _pendingRecorder: MediaRecorder | null = null;
   720→
   721→  stopVoiceCapture() {
   722→    if (this._pendingRecorder && this._pendingRecorder.state === 'recording') {
   723→      this._pendingRecorder.stop();
   724→    }
   725→  }

   727→  async visionSnapshot() {
   728→    if (!this.world?.switches.camera) {
   729→      this.error = 'Optical Camera Switch (#3) is OFF. Enable CAM to authorise optical capture.';
   730→      this.emit();
   731→      setTimeout(() => { this.error = null; this.emit(); }, 4200);

   735→    await this.sendConversation(
   736→      'Inspect current live optical camera frame at Vision Station.',
   737→      'vision'
   738→    );
   739→  }

   742→function blobToBase64(blob: Blob): Promise<string> {
   743→  return new Promise((resolve, reject) => {
   744→    const reader = new FileReader();
   745→    reader.onloadend = () => {
   746→      const result = String(reader.result || '');
   747→      resolve(result.split(',')[1] || '');
   748→    };
   749→    reader.onerror = reject;
   750→    reader.readAsDataURL(blob);
   751→  });
   752→}
