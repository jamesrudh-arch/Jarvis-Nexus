   405→  /** Cable conduits, hanging zone signage, etched floor plates. */
   406→  _buildZoneInfrastructure() {
   407→    const conduitMat = new THREE.MeshStandardMaterial({ color: 0x0c141b, roughness: 0.5, metalness: 0.7 });
   408→
   409→    // Floor cable conduits from the central ring to each station
   410→    for (const [tx, tz] of CONDUIT_TARGETS) {
   411→      const dir = new THREE.Vector2(tx, tz).normalize();
   412→      const start = new THREE.Vector3(dir.x * 2.1, 0.035, dir.y * 2.1);
   413→      const end = new THREE.Vector3(tx, 0.035, tz);
   414→      const mid = start.clone().add(end).multiplyScalar(0.5);
   415→      mid.y = 0.052;
   416→      const curve = new THREE.CatmullRomCurve3([start, mid, end]);
   417→      const tube = new THREE.Mesh(new THREE.TubeGeometry(curve, 28, 0.028, 6), conduitMat);
   418→      this.architectureGroup.add(tube);
   419→    }
   420→
   421→    // Hanging zone signage plates — two back-to-back front-face planes so the
   422→    // text reads correctly from BOTH sides (never mirrored)
   423→    for (const z of ZONE_SIGNAGE) {
   424→      if (!z.pos) continue;
   425→      const tex = createZoneLabelTexture(z.idx, z.title, z.sub, (z.idx === 8 || z.idx === 7) ? '#e7ae5b' : '#62e6f2');
   426→      const plateMat = new THREE.MeshBasicMaterial({ map: tex, transparent: true, depthWrite: false });
   427→      const plateGeo = new THREE.PlaneGeometry(1.62, 0.6);
   428→      const plateGroup = new THREE.Group();
   429→      plateGroup.position.set(z.pos[0], z.pos[1], z.pos[2]);
   430→      plateGroup.lookAt(0, z.pos[1], 0);
   431→      const front = new THREE.Mesh(plateGeo, plateMat);
   432→      front.position.z = 0.0015;
   433→      const back = new THREE.Mesh(plateGeo, plateMat);

   940→  renderFrame(dt, opticalVideoEl, cameraSwitchOn) {
   941→    // 1. Procedural video & volumetric idle elements
   942→    const scenarioActive = Boolean(this.scenarioDirector.activeScenario);
   943→    this.explainerSidePanel.visible = scenarioActive && this.mediaScreenMat.map === this.schlierenFeed.texture;
   944→    this.explainerSidePanelEdge.visible = this.explainerSidePanel.visible;
   945→    if (scenarioActive && this.mediaScreenMat.map === this.schlierenFeed.texture) {
   946→      // Documentary explainer feed synchronized to the active 3D scenario
   947→      this.schlierenFeed.mode = 'explainer';
   948→      this.explainerTimer += dt;
   949→      if (this.explainerTimer > 0.1) {
   950→        this.explainerTimer = 0;
   951→        this.schlierenFeed.renderExplainer(this.scenarioDirector.getMediaFeedState());
   952→      }
   953→    } else {
   954→      if (this.schlierenFeed.mode === 'explainer') {
   955→        this.schlierenFeed.mode = 'schlieren';
   956→        this.schlierenFeed.renderFrame(this.schlierenFeed.currentTime);
   957→      }
   958→      this.schlierenFeed.update(dt);
   959→    }
   960→    if (cameraSwitchOn && opticalVideoEl) {
   961→      this.updateVisionBoard(true, opticalVideoEl);
   962→    }
   963→
   964→    const holoClock = (this.holoClock = (this.holoClock || 0) + dt);
   965→    for (const m of this.centerHoloMats) m.uniforms.uTime.value = holoClock;
   966→    for (const m of this.idleHoloMats) m.uniforms.uTime.value = holoClock;
   967→
   968→    if (this.centerHoloAssembly) {
   969→      const stageActive = Boolean(this.scenarioDirector.activeScenario);
   970→      this.centerHoloAssembly.visible = !stageActive;
   971→      if (!this.reducedMotion) {
   972→        this.centerHoloAssembly.rotation.y += dt * 0.28;
   973→        if (this.centerOrbits) {
   974→          this.centerOrbits[0].rotation.z += dt * 0.55;
   975→          this.centerOrbits[1].rotation.z -= dt * 0.4;
   976→        }
   977→      }
   978→    }
   979→
   980→    if (!this.reducedMotion) {
   981→      if (this.securityLattice) this.securityLattice.rotation.y += dt * 0.22;
   982→      this.agentSolids.forEach((s, i) => {
   983→        s.rotation.x += dt * (0.35 + i * 0.08);
   984→        s.rotation.y += dt * 0.45;
   985→      });
   986→      if (this.extracted3DTwinGroup?.visible) {
   987→        this.extracted3DTwinGroup.rotation.y += dt * 0.65;
   988→      }
   989→      // Memory constellation breathing
   990→      for (const node of this.memoryNodes) {
   991→        const k = 1 + 0.35 * Math.sin(holoClock * 1.6 + node.userData.pulsePhase);
   992→        node.scale.setScalar(k);
   993→      }
   994→      // Dust motes drifting through the downlight cone
   995→      if (this.dustField) {
   996→        const pos = this.dustField.geometry.attributes.position;
   997→        for (let i = 0; i < pos.count; i++) {
   998→          let y = pos.getY(i) + dt * (0.045 + 0.02 * Math.sin(this.dustSeeds[i]));
   999→          let x = pos.getX(i) + Math.sin(holoClock * 0.35 + this.dustSeeds[i]) * dt * 0.05;
  1000→          if (y > 5.4) y = 0.25;
  1001→          pos.setY(i, y);
  1002→          pos.setX(i, x);
  1003→        }
  1004→        pos.needsUpdate = true;
  1005→      }
  1006→    }
  1007→
  1008→    // 2. Update Scenario Director, Living JARVIS, and Camera Director
  1009→    this.scenarioDirector.update(dt);
  1010→    this.jarvis.update(dt, this.camera.position, this.reducedMotion);
  1011→    this.cameraDirector.update(dt, this.jarvis, this.reducedMotion);
  1012→
  1013→    // 3. Project 3D world-space scenario callout tags onto screen
  1014→    this._renderSpatialAnchors();
  1015→
  1016→    // 4. Render either standard ACES Filmic sanctuary OR 4-Face Holographic Pyramid Scissor Viewports
  1017→    if (this.hologramMode && this.hologramLayout === 'pyramid_4face') {
  1018→      this._renderFourFacePyramid();
  1019→    } else {
  1020→      this.renderer.setScissorTest(false);
  1021→      this.composer.render();
  1022→    }

  1025→  _renderFourFacePyramid() {
  1026→    const w = window.innerWidth;
  1027→    const h = window.innerHeight;
  1028→    const size = Math.floor(Math.min(w, h) * 0.44);
  1029→    const cx = Math.floor(w / 2);
  1030→    const cy = Math.floor(h / 2);
  1031→
  1032→    this.renderer.setScissorTest(false);
  1033→    this.renderer.setClearColor(0x000000, 1);
  1034→    this.renderer.clear();
  1035→    this.renderer.setScissorTest(true);
  1036→
  1037→    const jPos = this.jarvis.root.position;
  1038→    const views = [
  1039→      { x: cx - size / 2, y: cy + 16, yaw: 0 },               // Top (0 deg)
  1040→      { x: cx + 16, y: cy - size / 2, yaw: Math.PI / 2 },     // Right (90 deg)
  1041→      { x: cx - size / 2, y: cy - size - 16, yaw: Math.PI },  // Bottom (180 deg)
  1042→      { x: cx - size - 16, y: cy - size / 2, yaw: -Math.PI / 2 }, // Left (270 deg)
  1043→    ];
  1044→
  1045→    for (const v of views) {
  1046→      const vx = Math.max(0, Math.floor(v.x));
  1047→      const vy = Math.max(0, Math.floor(v.y));
  1048→      const vw = Math.min(size, w - vx);
  1049→      const vh = Math.min(size, h - vy);
  1050→      if (vw <= 10 || vh <= 10) continue;
  1051→
  1052→      this.renderer.setViewport(vx, vy, vw, vh);
  1053→      this.renderer.setScissor(vx, vy, vw, vh);
  1054→      this.pyramidCam.aspect = vw / vh;
  1055→      this.pyramidCam.position.set(
  1056→        jPos.x + Math.sin(v.yaw) * 2.85,
  1057→        1.15,
  1058→        jPos.z + Math.cos(v.yaw) * 2.85
  1059→      );
  1060→      this.pyramidCam.lookAt(jPos.x, 0.98, jPos.z);
  1061→      this.pyramidCam.updateProjectionMatrix();
  1062→      this.renderer.render(this.scene, this.pyramidCam);
  1063→    }
  1064→    this.renderer.setViewport(0, 0, w, h);
  1065→    this.renderer.setScissorTest(false);
  1066→  }

  1068→  /** Project world anchors to screen with leader dots and simple collision offsets. */
  1069→  _renderSpatialAnchors() {
  1070→    const anchors = this.scenarioDirector.worldAnchors;
  1071→    if (!anchors || anchors.length === 0 || this.hologramMode) {
  1072→      if (this.labelsContainer.childElementCount > 0) this.labelsContainer.innerHTML = '';
  1073→      return;
  1074→    }
  1075→
  1076→    while (this.labelsContainer.childElementCount < anchors.length) {
  1077→      const div = document.createElement('div');
  1078→      div.className = 'spatial-tag';
  1079→      this.labelsContainer.appendChild(div);
  1080→    }
  1081→    while (this.labelsContainer.childElementCount > anchors.length) {
  1082→      this.labelsContainer.lastElementChild.remove();
  1083→    }
  1084→
  1085→    // Project & gather
  1086→    const placed = [];
  1087→    anchors.forEach((a, i) => {
  1088→      const el = this.labelsContainer.children[i];
  1089→      const proj = a.pos.clone().project(this.camera);
  1090→      if (proj.z > 1) {
  1091→        el.style.opacity = '0';
  1092→        return;
  1093→      }
  1094→      let sx = (proj.x * 0.5 + 0.5) * window.innerWidth;
  1095→      let sy = (-proj.y * 0.5 + 0.5) * window.innerHeight;
  1096→
  1097→      // Naive collision offset: push down until free of previously placed tags
  1098→      let guard = 0;
  1099→      let clash = placed.some((p) => Math.abs(p.sx - sx) < 300 && Math.abs(p.sy - sy) < 76);
  1100→      while (clash && guard < 6) {
  1101→        sy += 66;
  1102→        guard++;
  1103→        clash = placed.some((p) => Math.abs(p.sx - sx) < 300 && Math.abs(p.sy - sy) < 76);
  1104→      }
  1105→      placed.push({ sx, sy });
  1106→
  1107→      el.className = `spatial-tag ${a.amber ? 'amber' : ''}`;
  1108→      el.style.left = `${sx}px`;
  1109→      el.style.top = `${sy}px`;
  1110→      el.style.opacity = '1';
  1111→      el.innerHTML = `${a.title}<span class="tag-sub">${a.sub}</span>`;
  1112→    });
  1113→  }
