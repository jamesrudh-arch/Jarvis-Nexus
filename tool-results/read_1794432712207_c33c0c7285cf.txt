    99→    99→async function api<T = Record<string, unknown>>(path: string, options: RequestInit = {}): Promise<T> {
   100→   100→  const headers = options.body instanceof FormData ? {} : { 'Content-Type': 'application/json' };
   101→   101→  const res = await fetch(path, { ...options, headers: { ...headers, ...(options.headers as Record<string, string> | undefined) } });
   102→   102→  if (!res.ok) {
   103→   103→    let detail = `HTTP ${res.status}`;
   104→   104→    try {
   105→   105→      const err = await res.json();
   106→   106→      detail = (err as { detail?: string }).detail || detail;
   107→   107→    } catch { /* ignore */ }
   108→   108→    throw new Error(detail);
   109→   109→  }
   110→   110→  return (await res.json()) as T;
   111→   111→}

   191→   191→  async bootstrap() {
   192→   192→    this.emit();
   193→   193→    // Yield one tick so React StrictMode's double-mount cleanup lands before
   194→   194→    // the heavy 14 MB character download starts (prevents a duplicate load).
   195→   195→    await new Promise((r) => setTimeout(r, 0));
   196→   196→    const sanctuary = this.sanctuary;
   197→   197→    if (!sanctuary || this.disposed) return;
   198→   198→
   199→   199→    try {
   200→   200→      await sanctuary.initCharacter((xhr: { loaded: number; total: number } | undefined) => {
   201→   201→        if (this.disposed) return;
   202→   202→        if (xhr && xhr.total) {
   203→   203→          const pct = Math.round((xhr.loaded / xhr.total) * 100);
   204→   204→          this.loadingText = `Loading rigged 3D character & skeletal clips (${pct}%)…`;
   205→   205→          this.emit();
   206→   206→        }
   207→   207→      });
   208→   208→    } catch (err) {
   209→   209→      this.loadingText = `Character asset warning: ${(err as Error).message}`;
   210→   210→      this.emit();
   211→   211→    }
   212→   212→    if (this.disposed) return;
   213→   213→
   214→   214→    try {
   215→   215→      const [worldSnap, scenariosRes] = await Promise.all([
   216→   216→        api<WorldSnapshot>('/api/v1/world'),
   217→   217→        api<{ scenarios: ScenarioDef[] }>('/api/v1/scenarios'),
   218→   218→      ]);
   219→   219→      if (this.disposed) return;
   220→   220→      this.scenarios = scenariosRes.scenarios || [];
   221→   221→      this.applyWorldState(worldSnap, true);
   222→   222→    } catch (err) {
   223→   223→      this.error = (err as Error).message;
   224→   224→      this.emit();
   225→   225→      return;
   226→   226→    }
   227→   227→
   228→   228→    this.ready = true;
   229→   229→    this.loadingText = 'NEXUS online.';
   230→   230→    this.emit();
   231→   231→
   232→   232→    // Animation loop
   233→   233→    sanctuary.renderer.setAnimationLoop(() => {
   234→   234→      const dt = this.animClock.getDelta();
   235→   235→      sanctuary.renderFrame(dt, this.opticalCamEl, Boolean(this.world?.switches.camera));
   236→   236→      this.tickScenarioUI();
   237→   237→    });
   238→   238→
   239→   239→    // Telemetry poll
   240→   240→    this.telemetryTimer = setInterval(async () => {
   241→   241→      try {
   242→   242→        this.telemetry = await api<Telemetry>('/api/v1/telemetry');
   243→   243→        sanctuary.updateTelemetryBoard({
   244→   244→          ...this.telemetry,
   245→   245→          inference: this.telemetry.inference,
   246→   246→        } as never);
   247→   247→        this.emit();
   248→   248→      } catch { /* transient */ }
   249→   249→    }, 8000);
   250→   250→    void this.fetchTelemetryOnce();
   251→   251→  }

   253→   253→  private async fetchTelemetryOnce() {
   254→   254→    try {
   255→   255→      this.telemetry = await api<Telemetry>('/api/v1/telemetry');

   257→   257→      this.emit();
   258→   258→    } catch { /* ignore */ }
   259→   259→  }

   270→   270→  applyWorldState(snap: WorldSnapshot, initialSnap = false) {
   271→   271→    if (!snap || !this.sanctuary) return;
   272→   272→    this.world = snap;
   273→   273→    const sanctuary = this.sanctuary;
   274→   274→    sanctuary.reducedMotion = Boolean(snap.reduced_motion);
   275→   275→
   276→   276→    const zoneInfo = snap.zones?.[snap.current_zone];
   277→   277→    if (zoneInfo?.look_target) {
   278→   278→      sanctuary.jarvis.lookWorldTarget.set(...(zoneInfo.look_target as [number, number, number]));
   279→   279→    }
   280→   280→    sanctuary.jarvis.focusCamera = ['conversational_focus', 'wake_auth', 'digital_human_embodiment', 'holographic_presence'].includes(snap.active_mode);
   281→   281→
   282→   282→    const postureMap: Record<string, string> = {
   283→   283→      simulation_stage: snap.active_scenario ? 'scenario_direct' : 'scenario_construct',
   284→   284→      media_presentation: 'video_present',
   285→   285→      central_operations: snap.active_mode === 'conversational_focus' ? 'conversational_focus' : 'console_interaction',
   286→   286→    };
   287→   287→    const posture = postureMap[snap.current_zone] || 'console_interaction';
   288→   288→    const pos = snap.position;
   289→   289→
   290→   290→    // Autonomous behavior permission: only in ambient modes, autonomy switch ON,
   291→   291→    // no active scenario, and no pending conversation. She lives on her own.
   292→   292→    const ambientModes = ['living_presence', 'wake_auth', 'digital_human_embodiment'];
   293→   293→    sanctuary.behaviorDirector?.setServerPermitted(
   294→   294→      Boolean(snap.switches?.autonomy)
   295→   295→      && !snap.active_scenario
   296→   296→      && ambientModes.includes(snap.active_mode)
   297→   297→    );
   298→   298→
   299→   299→    if (initialSnap) {
   300→   300→      sanctuary.jarvis.root.position.set(pos.x, 0, pos.z);
   301→   301→      sanctuary.jarvis.root.rotation.y = pos.yaw;
   302→   302→      sanctuary.jarvis.targetYaw = pos.yaw;
   303→   303→      sanctuary.jarvis.postureMode = posture;
   304→   304→      sanctuary.jarvis.emotionalState = snap.emotional_state;
   305→   305→    } else {
   306→   306→      // Presenter's mark: while a scenario plays she stands BESIDE the hologram
   307→   307→      // (never inside the emitter volume), framed with the evidence she directs.
   308→   308→      let navX = pos.x;
   309→   309→      let navZ = pos.z;
   310→   310→      let navYaw = pos.yaw;
   311→   311→      if (snap.active_scenario && snap.current_zone === 'simulation_stage') {
   312→   312→        navX = 1.9;
   313→   313→        navZ = 2.9;
   314→   314→        navYaw = -2.11;
   315→   315→      }
   316→   316→      sanctuary.jarvis.navigateTo(navX, navZ, navYaw, {
   317→   317→        urgent: snap.locomotion_state === 'running' || snap.emotional_state === 'urgent',
   318→   318→        posture,
   319→   319→        emotion: snap.emotional_state || 'focused',
   320→   320→      });
   321→   321→    }
   322→   322→
   323→   323→    const isHolo = snap.active_mode === 'holographic_presence';
   324→   324→    sanctuary.setHologramMode(isHolo, this.holoLayout);
   325→   325→
   326→   326→    if (snap.active_camera_preset) {
   327→   327→      sanctuary.cameraDirector.setPreset(snap.active_camera_preset, sanctuary.reducedMotion);
   328→   328→    }
   329→   329→
   330→   330→    if (snap.active_scenario && !sanctuary.scenarioDirector.activeScenario) {
   331→   331→      const def = this.scenarios.find((s) => s.id === snap.active_scenario.scenario_id);
   332→   332→      if (def) this.activateClientScenario(def, snap.active_scenario);
   333→   333→    } else if (!snap.active_scenario && sanctuary.scenarioDirector.activeScenario) {
   334→   334→      sanctuary.scenarioDirector.clearStage();
   335→   335→      this.activeScenarioDef = null;
   336→   336→    }
   337→   337→
   338→   338→    sanctuary.syncPhysicalSwitchLevers(snap.switches || {});
   339→   339→    void this.enforceBrowserSensorGates();
   340→   340→    this.emit();
   341→   341→  }

   360→   360→      this.showJarvisResponse(payload.jarvis_message);

   362→   362→      if (payload.activated_scenario) {

   364→   364→        if (def) this.activateClientScenario(def, payload.activated_scenario);
   365→   365→      }
   366→   366→    } catch (err) {
   367→   367→      this.error = (err as Error).message;
   368→   368→      this.emit();
   369→   369→      setTimeout(() => { this.error = null; this.emit(); }, 4200);
   370→   370→    }

   607→   607→  private tickScenarioUI() {
   608→   608→    const sd = this.sanctuary?.scenarioDirector;
   609→   609→    if (!sd || !sd.activeScenario) {
   610→   610→      if (this.scenarioTimeReadout) {
   611→   611→        this.scenarioTimeReadout = '';
   612→   612→        this.scenarioPhase = '';
   613→   613→        this.emit();
   614→   614→      }
   615→   615→      this.lastNarrCueIndex = -1;
   616→   616→      return;
   617→   617→    }
   618→   618→
   619→   619→    // Scenario narration: when the timeline crosses into a new cue she SPEAKS
   620→   620→    // the phase narration with her synthesized voice + synced mouth, pointing
   621→   621→    // at the hologram. Skipped while she's already speaking (never overlaps).
   622→   622→    const cueIdx = sd.lastCueIndex;
   623→   623→    if (sd.playing && cueIdx !== this.lastNarrCueIndex && !this.speaking) {
   624→   624→      this.lastNarrCueIndex = cueIdx;
   625→   625→      const cue = sd.activeScenario.timeline?.[Math.max(0, cueIdx)];
   626→   626→      if (cue?.narration && this.narratingScenario) {
   627→   627→        void this.speakScenarioNarration(cue.narration);
   628→   628→      }
   629→   629→    }
   630→   630→
   631→   631→    const now = performance.now();
   632→   632→    if (now - this.lastTickEmit < 200) return;
   633→   633→    this.lastTickEmit = now;
   634→   634→
   635→   635→    this.scenarioTimeReadout = `${sd.currentTime.toFixed(1).padStart(4, '0')}s / ${sd.duration.toFixed(1)}s`;
   636→   636→    const tl = sd.activeScenario.timeline || [];
   637→   637→    let activeCue = tl[0];
   638→   638→    for (const cue of tl) {
   639→   639→      if (sd.currentTime >= cue.t) activeCue = cue;
   640→   640→    }
   641→   641→    if (activeCue) this.scenarioPhase = activeCue.phase;
   642→   642→    this.emit();
   643→   643→  }

   648→   648→    const sanctuary = this.sanctuary;

   657→   657→      sanctuary.jarvis.speak(text, {
   658→   658→        audio: voice?.audio ? { data: voice.audio, mime: voice.mime || 'audio/wav' } : undefined,
   659→   659→        onEnd: () => {
   660→   660→          this.narratingScenario = Boolean(sanctuary.scenarioDirector.activeScenario);
   661→   661→        },
   662→   662→      });
   663→   663→    } catch {
   664→   664→      if (sanctuary.scenarioDirector.activeScenario) {
   665→   665→        sanctuary.jarvis.speak(text);
   666→   666→      }
   667→   667→    }

   677→   677→  async enforceBrowserSensorGates() {
   678→   678→    if (!this.sanctuary || !this.world) return;
   679→   679→    const sw = this.world.switches;
   680→   680→
   681→   681→    if (!sw.microphone || !sw.master) {
   682→   682→      if (this.micStream) {
   683→   683→        this.micStream.getTracks().forEach((t) => t.stop());
   684→   684→        this.micStream = null;
   685→   685→      }
   686→   686→    }
   687→   687→
   688→   688→    if (!sw.camera || !sw.master) {
   689→   689→      if (this.camStream) {
   690→   690→        this.camStream.getTracks().forEach((t) => t.stop());
   691→   691→        this.camStream = null;
   692→   692→        if (this.opticalCamEl) this.opticalCamEl.srcObject = null;
   693→   693→        this.camState = 'off';
   694→   694→      }
   695→   695→      this.sanctuary.updateVisionBoard(false, null);
   696→   696→    } else if (!this.camStream && navigator.mediaDevices?.getUserMedia) {
   697→   697→      try {
   698→   698→        this.camStream = await navigator.mediaDevices.getUserMedia({ video: { width: 640, height: 360 } });
   699→   699→        if (this.opticalCamEl) {
   700→   700→          this.opticalCamEl.srcObject = this.camStream;
   701→   701→          await this.opticalCamEl.play().catch(() => undefined);
   702→   702→        }
   703→   703→        this.sanctuary.updateVisionBoard(true, this.opticalCamEl);
   704→   704→        this.camState = 'live';
   705→   705→      } catch {
   706→   706→        this.sanctuary.updateVisionBoard(false, null);
   707→   707→        this.camState = 'off';
   708→   708→      }
   709→   709→    }
   710→   710→    this.emit();
   711→   711→  }

   714→   714→  // Voice input: SpeechRecognition preferred, MediaRecorder+ASR fallback

   716→   716→  async startVoiceCapture(): Promise<void> {
   717→   717→    if (!this.world?.switches.microphone) {
   718→   718→      this.error = 'Microphone Switch (#2) is OFF. Enable MIC in the top switch bar to authorise audio capture.';
   719→   719→      this.emit();
   720→   720→      setTimeout(() => { this.error = null; this.emit(); }, 4200);
   721→   721→      return;
   722→   722→    }
   723→   723→
   724→   724→    const SR = (window as unknown as { SpeechRecognition?: new () => SpeechRecognition; webkitSpeechRecognition?: new () => SpeechRecognition }).SpeechRecognition
   725→   725→      ?? (window as unknown as { webkitSpeechRecognition?: new () => SpeechRecognition }).webkitSpeechRecognition;
   726→   726→
   727→   727→    if (SR) {
   728→   728→      const rec = new SR();
   729→   729→      rec.lang = 'en-US';
   730→   730→      rec.interimResults = false;
   731→   731→      this.micState = 'listening';
   732→   732→      this.emit();
   733→   733→      rec.onresult = async (ev: SpeechRecognitionEvent) => {
   734→   734→        const transcript = ev.results[0][0].transcript;
   735→   735→        this.micState = 'processing';
   736→   736→        this.emit();
   737→   737→        await this.sendConversation(transcript, 'voice');
   738→   738→        this.micState = 'off';
   739→   739→        this.emit();
   740→   740→      };
   741→   741→      rec.onerror = () => {
   742→   742→        this.micState = 'off';
   743→   743→        this.emit();
   744→   744→      };
   745→   745→      rec.onend = () => {
   746→   746→        if (this.micState === 'listening') {
   747→   747→          this.micState = 'off';
   748→   748→          this.emit();
   749→   749→        }
   750→   750→      };
   751→   751→      try { rec.start(); } catch { /* already started */ }
   752→   752→      return;
   753→   753→    }
   754→   754→
   755→   755→    // Fallback: record webm/opus, transcribe via CORE ASR
   756→   756→    if (!navigator.mediaDevices?.getUserMedia || typeof MediaRecorder === 'undefined') {
   757→   757→      this.error = 'Voice capture is unavailable in this browser. Type directly instead.';
   758→   758→      this.emit();
   759→   759→      setTimeout(() => { this.error = null; this.emit(); }, 4200);
   760→   760→      return;
   761→   761→    }
   762→   762→    try {
   763→   763→      this.micStream = await navigator.mediaDevices.getUserMedia({ audio: true });
   764→   764→      const recorder = new MediaRecorder(this.micStream);
   765→   765→      const chunks: Blob[] = [];
   766→   766→      this.micState = 'listening';
   767→   767→      this.emit();
   768→   768→      recorder.ondataavailable = (ev) => { if (ev.data.size > 0) chunks.push(ev.data); };
   769→   769→      recorder.onstop = async () => {
   770→   770→        this.micStream?.getTracks().forEach((t) => t.stop());
   771→   771→        this.micState = 'processing';
   772→   772→        this.emit();
   773→   773→        try {
   774→   774→          const blob = new Blob(chunks, { type: chunks[0]?.type || 'audio/webm' });
   775→   775→          const b64 = await blobToBase64(blob);
   776→   776→          const res = await api<{ text: string }>('/api/v1/asr', {
   777→   777→            method: 'POST',
   778→   778→            body: JSON.stringify({ audio_base64: b64 }),
   779→   779→          });
   780→   780→          if (res.text) {
   781→   781→            await this.sendConversation(res.text, 'voice');
   782→   782→          } else {
   783→   783→            this.error = 'No speech detected.';
   784→   784→            this.emit();
   785→   785→            setTimeout(() => { this.error = null; this.emit(); }, 3000);
   786→   786→          }
   787→   787→        } catch (err) {
   788→   788→          this.error = `Voice transcription failed: ${(err as Error).message}`;
   789→   789→          this.emit();
   790→   790→          setTimeout(() => { this.error = null; this.emit(); }, 4200);
   791→   791→        }
   792→   792→        this.micState = 'off';
   793→   793→        this.emit();
   794→   794→      };
   795→   795→      recorder.start();
   796→   796→      // Auto-stop after 12s safety window
   797→   797→      setTimeout(() => {
   798→   798→        if (recorder.state === 'recording') recorder.stop();
   799→   799→      }, 12000);
   800→   800→      this._pendingRecorder = recorder;
   801→   801→    } catch {
   802→   802→      this.micState = 'off';
   803→   803→      this.error = 'Microphone permission denied.';
   804→   804→      this.emit();
   805→   805→      setTimeout(() => { this.error = null; this.emit(); }, 3600);
   806→   806→    }
   807→   807→  }

   832→   832→function blobToBase64(blob: Blob): Promise<string> {
   833→   833→  return new Promise((resolve, reject) => {
   834→   834→    const reader = new FileReader();
   835→   835→    reader.onloadend = () => {
   836→   836→      const result = String(reader.result || '');
   837→   837→      resolve(result.split(',')[1] || '');
   838→   838→    };
   839→   839→    reader.onerror = reject;
   840→   840→    reader.readAsDataURL(blob);
   841→   841→  });
   842→   842→}
