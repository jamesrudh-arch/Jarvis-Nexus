    11→let MAX_ANISO = 8;
    12→export function setGlobalAnisotropy(n) {
    13→  MAX_ANISO = Math.max(1, Math.min(16, n | 0));
    14→}
    15→function applyAniso(tex) {
    16→  tex.anisotropy = MAX_ANISO;
    17→  tex.needsUpdate = true;
    18→  return tex;
    19→}

    21→function mulberry32(seed) {
    22→  let a = seed >>> 0;
    23→  return function () {
    24→    a |= 0; a = (a + 0x6d2b79f5) | 0;
    25→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    26→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    27→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    28→  };
    29→}

   170→/**
   171→ * Tiling micro-surface normal map generated from value noise (finite-difference
   172→ * bump → normal). Gives skin pore-level and fabric weave specular breakup so
   173→ * close-ups never read as airbrushed wax or flat cloth.
   174→ */
   175→export function createMicroNormalTexture({ size = 256, scale = 24, strength = 1.0, weave = false, seed = 1337 } = {}) {
   176→  const height = document.createElement('canvas');
   177→  height.width = height.height = size;
   178→  const hCtx = height.getContext('2d');
   179→  const img = hCtx.createImageData(size, size);
   180→  const rand = mulberry32(seed);
   181→
   182→  // Multi-octave value noise field
   183→  const gridN = Math.max(2, Math.round(scale));
   184→  const grids = [];
   185→  for (let o = 0; o < 3; o++) {
   186→    const n = gridN * (1 << o);
   187→    const g = new Float32Array((n + 1) * (n + 1));
   188→    for (let i = 0; i < g.length; i++) g[i] = rand();
   189→    grids.push({ n, g });
   190→  }
   191→  const sample = (x, y) => {
   192→    let v = 0, amp = 1, tot = 0;
   193→    for (const { n, g } of grids) {
   194→      const fx = x * n, fy = y * n;
   195→      const x0 = Math.floor(fx), y0 = Math.floor(fy);
   196→      const tx = fx - x0, ty = fy - y0;
   197→      const sx = tx * tx * (3 - 2 * tx), sy = ty * ty * (3 - 2 * ty);
   198→      const i00 = g[(y0 % (n + 1)) * (n + 1) + (x0 % (n + 1))];
   199→      const i10 = g[(y0 % (n + 1)) * (n + 1) + ((x0 + 1) % (n + 1))];
   200→      const i01 = g[((y0 + 1) % (n + 1)) * (n + 1) + (x0 % (n + 1))];
   201→      const i11 = g[((y0 + 1) % (n + 1)) * (n + 1) + ((x0 + 1) % (n + 1))];
   202→      v += (i00 * (1 - sx) + i10 * sx) * (1 - sy) * amp + (i01 * (1 - sx) + i11 * sx) * sy * amp;
   203→      tot += amp;
   204→      amp *= 0.5;
   205→    }
   206→    return v / tot;
   207→  };
   208→
   209→  for (let y = 0; y < size; y++) {
   210→    for (let x = 0; x < size; x++) {
   211→      const u = x / size, v = y / size;
   212→      let h = sample(u, v);
   213→      if (weave) {
   214→        // crossed warp/weft threads
   215→        const warp = Math.sin(u * Math.PI * 2 * 46) * 0.5 + 0.5;
   216→        const weft = Math.sin(v * Math.PI * 2 * 46) * 0.5 + 0.5;
   217→        h = h * 0.45 + Math.max(warp, weft) * 0.4 + Math.min(warp, weft) * 0.15;
   218→      }
   219→      img.data[(y * size + x) * 4] = img.data[(y * size + x) * 4 + 1] = img.data[(y * size + x) * 4 + 2] = h * 255;
   220→      img.data[(y * size + x) * 4 + 3] = 255;
   221→    }
   222→  }
   223→  hCtx.putImageData(img, 0, 0);
   224→
   225→  // Finite-difference → normal map (wrap-around for tiling)
   226→  const out = document.createElement('canvas');
   227→  out.width = out.height = size;
   228→  const oCtx = out.getContext('2d');
   229→  const oImg = oCtx.createImageData(size, size);
   230→  const lum = (x, y) => img.data[(((y + size) % size) * size + ((x + size) % size)) * 4] / 255;
   231→  for (let y = 0; y < size; y++) {
   232→    for (let x = 0; x < size; x++) {
   233→      const dx = (lum(x + 1, y) - lum(x - 1, y)) * strength;
   234→      const dy = (lum(x, y + 1) - lum(x, y - 1)) * strength;
   235→      const inv = 1 / Math.sqrt(dx * dx + dy * dy + 1);
   236→      const i = (y * size + x) * 4;
   237→      oImg.data[i] = (-dx * inv * 0.5 + 0.5) * 255;
   238→      oImg.data[i + 1] = (-dy * inv * 0.5 + 0.5) * 255;
   239→      oImg.data[i + 2] = inv * 255;
   240→      oImg.data[i + 3] = 255;
   241→    }
   242→  }
   243→  oCtx.putImageData(oImg, 0, 0);
   244→  const tex = new THREE.CanvasTexture(out);
   245→  tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
   246→  applyAniso(tex);
   247→  return tex;
   248→}

   322→/**
   323→ * Remaps bright white inner-blouse pixels on Avaturn's suit texture to a tailored
   324→ * midnight-slate knit (#0E1822) and deepens suit charcoal (#1C252E) to match our approved storyboard.
   325→ */
   326→export function tailorSuitAlbedoTexture(sourceTex) {
   327→  if (!sourceTex || !sourceTex.image) return sourceTex;
   328→  const img = sourceTex.image;
   329→  const w = img.width || 1024;
   330→  const h = img.height || 1024;
   331→  const canvas = document.createElement('canvas');
   332→  canvas.width = w;
   333→  canvas.height = h;
   334→  const ctx = canvas.getContext('2d');
   335→  ctx.drawImage(img, 0, 0, w, h);
   336→
   337→  const imgData = ctx.getImageData(0, 0, w, h);
   338→  const d = imgData.data;
   339→  for (let i = 0; i < d.length; i += 4) {
   340→    const r = d[i];
   341→    const g = d[i + 1];
   342→    const b = d[i + 2];
   343→    const lum = 0.299 * r + 0.587 * g + 0.114 * b;
   344→    const sat = Math.max(r, g, b) - Math.min(r, g, b);
   345→
   346→    if (lum > 145 && sat < 38) {
   347→      // Inner white shirt -> deep midnight mock-neck knit (#0E1822) preserving fold shading
   348→      const shade = lum / 255;
   349→      d[i] = Math.round(14 * shade);
   350→      d[i + 1] = Math.round(25 * shade);
   351→      d[i + 2] = Math.round(36 * shade);
   352→    } else {
   353→      // Tailor grey suit -> rich architectural charcoal-slate (#1D2630)
   354→      const factor = 0.62;
   355→      d[i] = Math.min(255, Math.round(r * factor * 0.92));
   356→      d[i + 1] = Math.min(255, Math.round(g * factor * 0.98));
   357→      d[i + 2] = Math.min(255, Math.round(b * factor * 1.08));
   358→    }
   359→  }
   360→  ctx.putImageData(imgData, 0, 0);
   361→  const out = new THREE.CanvasTexture(canvas);
   362→  out.flipY = sourceTex.flipY;
   363→  out.colorSpace = THREE.SRGBColorSpace;
   364→  applyAniso(out);
   365→  out.needsUpdate = true;
   366→  return out;
   367→}

   369→/**
   370→ * Real-time procedural feeds for the frameless Media Wall.
   371→ *  - Schlieren optical stator benchmark (default idle clip)
   372→ *  - Explainer mode: a documentary-style annotated panel synced live to the
   373→ *    active 3D scenario (phase captions, cue timeline, measured curves,
   374→ *    live variable readouts, verified-fact vs inference legend).
   375→ */
   376→export class SchlierenFeedRenderer {
   377→  constructor() {
   378→    this.canvas = document.createElement('canvas');
   379→    this.canvas.width = 1280;
   380→    this.canvas.height = 720;
   381→    this.ctx = this.canvas.getContext('2d');
   382→    this.texture = new THREE.CanvasTexture(this.canvas);
   383→    this.texture.colorSpace = THREE.SRGBColorSpace;
   384→    applyAniso(this.texture);
   385→    this.currentTime = 14.2;
   386→    this.playing = false;
   387→    this.duration = 24.0;
   388→    this.mode = 'schlieren';
   389→    this.renderFrame(14.2);
   390→  }
   391→
   392→  update(dt) {
   393→    if (this.playing) {
   394→      this.currentTime = (this.currentTime + dt) % this.duration;
   395→      this.renderFrame(this.currentTime);
   396→    }
   397→  }
   398→
   399→  seek(t) {
   400→    this.currentTime = Math.max(0, Math.min(this.duration, t));
   401→    this.renderFrame(this.currentTime);
   402→  }
   403→
   404→  renderFrame(t) {
   405→    if (this.mode === 'explainer') return; // drawn per-state, not per-clock
   406→    const { ctx, canvas } = this;
   407→    const w = canvas.width;
   408→    const h = canvas.height;
   409→
   410→    // High-contrast monochrome/ice Schlieren optical wind-tunnel field
   411→    const grad = ctx.createLinearGradient(0, 0, w, h);
   412→    grad.addColorStop(0, '#071017');
   413→    grad.addColorStop(0.5, '#172733');
   414→    grad.addColorStop(1, '#060d13');
   415→    ctx.fillStyle = grad;
   416→    ctx.fillRect(0, 0, w, h);
   417→
   418→    const cx = w * 0.48;
   419→    const cy = h * 0.5;
   420→    const outerR = 250;
   421→    const hubR = 96;
   422→
   423→    ctx.strokeStyle = '#c8e4ec';
   424→    ctx.lineWidth = 4;
   425→    ctx.beginPath(); ctx.arc(cx, cy, outerR, 0, Math.PI * 2); ctx.stroke();
   426→    ctx.beginPath(); ctx.arc(cx, cy, hubR, 0, Math.PI * 2); ctx.stroke();
   427→
   428→    const vortexSurge = Math.exp(-Math.pow(t - 14.2, 2) / 6.5);
   429→    for (let i = 0; i < 18; i++) {
   430→      const ang = (i / 18) * Math.PI * 2 + t * 0.65;
   431→      const x1 = cx + Math.cos(ang) * hubR;
   432→      const y1 = cy + Math.sin(ang) * hubR;
   433→      const x2 = cx + Math.cos(ang + 0.14) * outerR;
   434→      const y2 = cy + Math.sin(ang + 0.14) * outerR;
   435→
   436→      ctx.strokeStyle = 'rgba(221, 251, 255, 0.86)';
   437→      ctx.lineWidth = 3;
   438→      ctx.beginPath();
   439→      ctx.moveTo(x1, y1);
   440→      ctx.quadraticCurveTo(
   441→        cx + Math.cos(ang + 0.09) * ((hubR + outerR) * 0.52),
   442→        cy + Math.sin(ang + 0.09) * ((hubR + outerR) * 0.52),
   443→        x2, y2
   444→      );
   445→      ctx.stroke();
   446→
   447→      ctx.strokeStyle = vortexSurge > 0.45 && i % 3 === 0
   448→        ? `rgba(231, 174, 91, ${0.35 + 0.5 * vortexSurge})`
   449→        : 'rgba(98, 230, 242, 0.26)';
   450→      ctx.lineWidth = 1.8;
   451→      ctx.beginPath();
   452→      ctx.arc(cx, cy, hubR + 26 + (i % 5) * 24, ang, ang + 0.24 + vortexSurge * 0.22);
   453→      ctx.stroke();
   454→    }
   455→
   456→    if (vortexSurge > 0.35) {
   457→      const vx = cx + 122;
   458→      const vy = cy - 56;
   459→      ctx.strokeStyle = `rgba(231, 174, 91, ${vortexSurge})`;
   460→      ctx.lineWidth = 2.4;
   461→      ctx.strokeRect(vx - 48, vy - 48, 96, 96);

   482→  /**

   686→  _sampleSeries(s, t) {
   687→    if (!s.points || !s.points.length) return null;
   688→    if (t <= s.points[0].t) return s.points[0].v;
   689→    const last = s.points[s.points.length - 1];
   690→    if (t >= last.t) return last.v;
   691→    for (let i = 1; i < s.points.length; i++) {
   692→      if (s.points[i].t >= t) {
   693→        const a = s.points[i - 1], b = s.points[i];
   694→        const f = (t - a.t) / Math.max(0.001, b.t - a.t);
   695→        return a.v + (b.v - a.v) * f;
   696→      }
   697→    }
   698→    return last.v;
   699→  }
