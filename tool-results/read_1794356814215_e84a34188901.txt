    30→// Optional external CORE origin when Frontend is hosted separately (e.g. Vercel -> Railway).
    31→// Defaults to '' (same-origin relative URLs) when served directly by Railway / FastAPI / Vite proxy.
    32→const CORE_HTTP_ORIGIN = (import.meta.env.VITE_CORE_URL || '').trim().replace(/\/+$/, '');
    33→
    34→function coreHttpUrl(path) {
    35→  return CORE_HTTP_ORIGIN ? `${CORE_HTTP_ORIGIN}${path}` : path;
    36→}
    37→
    38→function coreWsBase() {
    39→  if (CORE_HTTP_ORIGIN) {
    40→    return CORE_HTTP_ORIGIN.replace(/^http:/i, 'ws:').replace(/^https:/i, 'wss:');
    41→  }
    42→  const proto = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
    43→  return `${proto}//${window.location.host}`;
    44→}
    45→
    46→async function api(path, options = {}) {
    47→  const headers = options.body instanceof FormData ? {} : { 'Content-Type': 'application/json' };
    48→  const res = await fetch(coreHttpUrl(path), {
    49→    ...options,
    50→    headers: { ...headers, ...(options.headers || {}) },
    51→  });
    52→  if (!res.ok) {
    53→    let detail = `HTTP ${res.status}`;
    54→    try {
    55→      const err = await res.json();
    56→      detail = err.detail || detail;
    57→    } catch { /* ignore */ }
    58→    throw new Error(detail);
    59→  }
    60→  return res.json();
    61→}

   106→function connectNexusWebSocket() {
   107→  const isMobile = window.innerWidth <= 860;
   108→  const deviceTag = isMobile ? 'mobile_companion' : 'pc_sanctuary';
   109→  const wsUrl = `${coreWsBase()}/ws/nexus?device=${deviceTag}`;
   110→
   111→  const dot = document.getElementById('core-conn-dot');
   112→  const ws = new WebSocket(wsUrl);
   113→  state.ws = ws;
   114→
   115→  ws.onopen = () => dot.classList.remove('offline');
   116→  ws.onclose = () => {
   117→    dot.classList.add('offline');
   118→    setTimeout(connectNexusWebSocket, 3500);
   119→  };
   120→  ws.onmessage = (ev) => {
   121→    try {
   122→      const msg = JSON.parse(ev.data);
   123→      handleLiveEvent(msg);
   124→    } catch { /* noop */ }
   125→  };
   126→}

   128→function handleLiveEvent(msg) {
   129→  const { type, payload } = msg;
   130→  if (type === 'initial_state' || type === 'world_sync' || type === 'scenario_restored') {
   131→    applyWorldState(payload, false);
   132→    if (type === 'scenario_restored') {
   133→      sanctuary.scenarioDirector.clearStage();
   134→      document.getElementById('scenario-director-bar').classList.add('hidden');
   135→    }
   136→  } else if (type === 'jarvis_interrupted') {
   137→    applyWorldState(payload, false);
   138→    sanctuary.jarvis.acknowledgeInterruption(sanctuary.camera.position);
   139→  } else if (type === 'switch_changed') {
   140→    applyWorldState(payload.world, false);
   141→    enforceBrowserSensorGates();
   142→  } else if (type === 'jarvis_spoke') {
   143→    applyWorldState(payload.world, false);
   144→    showJarvisResponse(payload.jarvis_message);
   145→    if (payload.activated_scenario) {
   146→      const def = state.scenarios.find((s) => s.id === payload.activated_scenario.scenario_id);
   147→      if (def) activateClientScenario(def, payload.activated_scenario);
   148→    }
   149→  } else if (type === 'scenario_activated') {
   150→    applyWorldState(payload.world, false);
   151→    activateClientScenario(payload.scenario, payload.runtime);
   152→  } else if (type === 'video_state_changed') {
   153→    applyWorldState(payload.world, false);
   154→    applyVideoState(payload.video);
   155→  }
   156→}

   161→function applyWorldState(snap, initialSnap = false) {
   162→  if (!snap) return;
   163→  state.world = snap;
   164→  state.switches = snap.switches || {};
   165→  sanctuary.reducedMotion = Boolean(snap.reduced_motion);
   166→  document.getElementById('chk-reduced-motion').checked = sanctuary.reducedMotion;
   167→
   168→  // Header status
   169→  const modeObj = (snap.modes || []).find((m) => m.id === snap.active_mode);
   170→  document.getElementById('active-mode-label').textContent = modeObj ? modeObj.label : snap.active_mode;
   171→  document.getElementById('jarvis-zone-badge').textContent = (snap.current_zone || '').replaceAll('_', ' ');
   172→  document.getElementById('jarvis-activity-text').textContent = snap.active_activity_summary || '';
   173→
   174→  // Resume button visibility when an interrupted task is on the stack
   175→  const hasStack = Array.isArray(snap.interrupted_task_stack) && snap.interrupted_task_stack.length > 0;
   176→  document.getElementById('btn-resume-task').classList.toggle('hidden', !hasStack);
   177→
   178→  // Render 7 Hardware Switches
   179→  renderSwitchesStrip(snap.switches || {});
   180→  sanctuary.syncPhysicalSwitchLevers(snap.switches || {});
   181→
   182→  // Render 15 Modes Drawer
   183→  renderModesDrawer(snap.modes || [], snap.active_mode);
   184→
   185→  // Move or place JARVIS in 3D space
   186→  const pos = snap.position || { x: 1.45, z: 0.95, yaw: -0.42 };
   187→  const zoneInfo = snap.zones?.[snap.current_zone];
   188→  if (zoneInfo?.look_target) {
   189→    sanctuary.jarvis.lookWorldTarget.set(...zoneInfo.look_target);
   190→  }
   191→  sanctuary.jarvis.focusCamera = ['conversational_focus', 'wake_auth', 'digital_human_embodiment', 'holographic_presence'].includes(snap.active_mode);
   192→
   193→  const postureMap = {
   194→    simulation_stage: snap.active_scenario ? 'scenario_direct' : 'scenario_construct',
   195→    media_presentation: 'video_present',
   196→    central_operations: snap.active_mode === 'conversational_focus' ? 'conversational_focus' : 'console_interaction',
   197→  };
   198→  const posture = postureMap[snap.current_zone] || 'console_interaction';
   199→
   200→  if (initialSnap) {
   201→    sanctuary.jarvis.root.position.set(pos.x, 0, pos.z);
   202→    sanctuary.jarvis.root.rotation.y = pos.yaw;
   203→    sanctuary.jarvis.targetYaw = pos.yaw;
   204→    sanctuary.jarvis.postureMode = posture;
   205→    sanctuary.jarvis.emotionalState = snap.emotional_state || 'attentive';
   206→  } else {
   207→    sanctuary.jarvis.navigateTo(pos.x, pos.z, pos.yaw, {
   208→      urgent: snap.locomotion_state === 'running' || snap.emotional_state === 'urgent',
   209→      posture,
   210→      emotion: snap.emotional_state || 'focused',
   211→    });
   212→  }
   213→
   214→  // Camera & Holographic Mode 15 sync
   215→  const isHolo = snap.active_mode === 'holographic_presence';
   216→  sanctuary.setHologramMode(isHolo, sanctuary.hologramLayout);
   217→  document.getElementById('hologram-control-bar').classList.toggle('hidden', !isHolo);
   218→  document.getElementById('hologram-guides').classList.toggle('hidden', !isHolo);
   219→  document.getElementById('btn-hologram-toggle').classList.toggle('active', isHolo);
   220→
   221→  if (snap.active_camera_preset) {
   222→    sanctuary.cameraDirector.setPreset(snap.active_camera_preset, sanctuary.reducedMotion);
   223→  }
   224→
   225→  if (snap.active_scenario && !sanctuary.scenarioDirector.activeScenario) {
   226→    const def = state.scenarios.find((s) => s.id === snap.active_scenario.scenario_id);
   227→    if (def) activateClientScenario(def, snap.active_scenario);
   228→  }
   229→
   230→  if (snap.active_video) {
   231→    applyVideoState(snap.active_video);
   232→  }
   233→}

   269→// ---------------------------------------------------------------------------
   270→// 3. Real Browser Sensor Enforcement (No Hidden Capture!)
   271→// ---------------------------------------------------------------------------
   272→async function enforceBrowserSensorGates() {
   273→  // Microphone enforcement
   274→  if (!state.switches.microphone || !state.switches.master) {
   275→    if (state.micStream) {
   276→      state.micStream.getTracks().forEach((t) => t.stop());
   277→      state.micStream = null;
   278→    }
   279→    if (state.recognition) {
   280→      try { state.recognition.abort(); } catch { /* noop */ }
   281→    }
   282→  }
   283→
   284→  // Camera enforcement
   285→  if (!state.switches.camera || !state.switches.master) {
   286→    if (state.camStream) {
   287→      state.camStream.getTracks().forEach((t) => t.stop());
   288→      state.camStream = null;
   289→      opticalCamEl.srcObject = null;
   290→    }
   291→    sanctuary.updateVisionBoard(false, null);
   292→  } else if (!state.camStream && navigator.mediaDevices?.getUserMedia) {
   293→    try {
   294→      state.camStream = await navigator.mediaDevices.getUserMedia({ video: { width: 640, height: 360 } });
   295→      opticalCamEl.srcObject = state.camStream;
   296→      await opticalCamEl.play();
   297→      sanctuary.updateVisionBoard(true, opticalCamEl);
   298→    } catch {
   299→      sanctuary.updateVisionBoard(false, null);
   300→    }
   301→  }
   302→}

   335→function updateScenarioBranchBanner(def, vars) {
   336→  const banner = document.getElementById('sc-branch-readout');
   337→  if (!def || !def.branches || def.branches.length === 0) {
   338→    banner.textContent = '';
   339→    return;
   340→  }
   341→  let chosen = def.branches[def.branches.length - 1];
   342→  if (def.id === 'cryo_manifold_forensic') {
   343→    const ruptured = Number(vars.line_pressure_mpa ?? 210) >= 185 && Number(vars.relief_valve_ms ?? 125) > 70;
   344→    chosen = ruptured ? def.branches[0] : def.branches[1];
   345→  } else if (def.id === 'dual_turbine_sim') {
   346→    chosen = Number(vars.inlet_temp_k ?? 1320) <= 1450 ? def.branches[0] : def.branches[1];
   347→  } else if (def.id === 'zerotrust_compute_flow') {
   348→    chosen = Number(vars.expose_public_port ?? 0) === 0 ? def.branches[0] : def.branches[1];
   349→  }
   350→  banner.className = `scenario-branch-banner ${chosen.severity === 'critical' ? 'critical' : ''}`;
   351→  banner.textContent = `${chosen.label} — ${chosen.outcome}`;
   352→}

   354→function syncScenarioSliderReadout() {
   355→  const sd = sanctuary.scenarioDirector;
   356→  if (!sd.activeScenario) return;
   357→  document.getElementById('sc-time-readout').textContent =
   358→    `${sd.currentTime.toFixed(1).padStart(4, '0')}s / ${sd.duration.toFixed(1)}s`;
   359→  if (sd.playing) {
   360→    document.getElementById('sc-timeline-slider').value = String(sd.currentTime);
   361→  }
   362→  const tl = sd.activeScenario.timeline || [];
   363→  let activeCue = tl[0];
   364→  for (const cue of tl) {
   365→    if (sd.currentTime >= cue.t) activeCue = cue;
   366→  }
   367→  if (activeCue) {
   368→    document.getElementById('sc-phase-label').textContent = activeCue.phase;
   369→  }
   370→}

   372→function applyVideoState(vState) {
   373→  if (!vState) return;
   374→  sanctuary.schlierenFeed.playing = Boolean(vState.playing);
   375→  if (!vState.playing && typeof vState.current_time === 'number') {
   376→    sanctuary.schlierenFeed.seek(vState.current_time);
   377→  }
   378→  if (sanctuary.extracted3DTwinGroup) {
   379→    sanctuary.extracted3DTwinGroup.visible = Boolean(vState.extracted_3d_twin);
   380→  }
   381→  if (!uploadedVideoEl.paused && !vState.playing) uploadedVideoEl.pause();
   382→  if (uploadedVideoEl.src && vState.playing && uploadedVideoEl.paused) {
   383→    uploadedVideoEl.play().catch(() => {});
   384→  }
   385→}

   875→async function handleVoiceCaptureClick() {
   876→  if (!state.switches.microphone) {
   877→    alert('Microphone Switch (#2) is currently OFF. Enable MIC in the top switch bar to authorise audio capture.');
   878→    return;
   879→  }
   880→  const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
   881→  if (!SR) {
   882→    alert('Browser Web Speech Recognition API is unavailable in this browser. Type directly or connect an audio worker.');
   883→    return;
   884→  }
   885→  const rec = new SR();
   886→  rec.lang = 'en-US';
   887→  rec.interimResults = false;
   888→  state.recognition = rec;
   889→  document.getElementById('mic-state-label').textContent = 'LISTENING...';
   890→  rec.onresult = async (ev) => {
   891→    const transcript = ev.results[0][0].transcript;
   892→    document.getElementById('mic-state-label').textContent = 'MIC READY';
   893→    await api('/api/v1/conversations/send', {
   894→      method: 'POST',
   895→      body: JSON.stringify({ content: transcript, modality: 'voice' }),
   896→    });
   897→  };
   898→  rec.onerror = () => {
   899→    document.getElementById('mic-state-label').textContent = 'MIC READY';
   900→  };
   901→  rec.start();
   902→}

   920→function toggleWebMScenarioRecording(e) {
   921→  const btn = e.target;
   922→  if (state.mediaRecorder && state.mediaRecorder.state === 'recording') {
   923→    state.mediaRecorder.stop();
   924→    btn.textContent = 'Rec WebM';
   925→    return;
   926→  }
   927→  if (!canvas.captureStream || typeof MediaRecorder === 'undefined') {
   928→    alert('MediaRecorder WebM capture is not supported by this browser.');
   929→    return;
   930→  }
   931→  const stream = canvas.captureStream(30);
   932→  const recorder = new MediaRecorder(stream, { mimeType: 'video/webm' });
   933→  state.recordedChunks = [];
   934→  recorder.ondataavailable = (ev) => {
   935→    if (ev.data.size > 0) state.recordedChunks.push(ev.data);
   936→  };
   937→  recorder.onstop = async () => {
   938→    const blob = new Blob(state.recordedChunks, { type: 'video/webm' });
   939→    const fd = new FormData();
   940→    fd.append('title', `Scenario Recording ${new Date().toISOString().slice(11, 19)}`);
   941→    fd.append('source_kind', 'scenario_recording');
   942→    fd.append('file', blob, 'scenario_capture.webm');
   943→    await api('/api/v1/media/upload', { method: 'POST', body: fd });
   944→    const url = URL.createObjectURL(blob);
   945→    const a = document.createElement('a');
   946→    a.href = url;
   947→    a.download = 'jarvis_nexus_scenario.webm';
   948→    a.click();
   949→  };
   950→  recorder.start();
   951→  state.mediaRecorder = recorder;
   952→  btn.textContent = 'Stop Rec';
   953→}
