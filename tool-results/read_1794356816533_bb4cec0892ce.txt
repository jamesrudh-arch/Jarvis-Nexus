    34→    34→function coreHttpUrl(path) {
    35→    35→  return CORE_HTTP_ORIGIN ? `${CORE_HTTP_ORIGIN}${path}` : path;
    36→    36→}

    38→    38→function coreWsBase() {
    39→    39→  if (CORE_HTTP_ORIGIN) {
    40→    40→    return CORE_HTTP_ORIGIN.replace(/^http:/i, 'ws:').replace(/^https:/i, 'wss:');
    41→    41→  }
    42→    42→  const proto = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
    43→    43→  return `${proto}//${window.location.host}`;
    44→    44→}

    46→    46→async function api(path, options = {}) {
    47→    47→  const headers = options.body instanceof FormData ? {} : { 'Content-Type': 'application/json' };
    48→    48→  const res = await fetch(coreHttpUrl(path), {
    49→    49→    ...options,
    50→    50→    headers: { ...headers, ...(options.headers || {}) },
    51→    51→  });
    52→    52→  if (!res.ok) {
    53→    53→    let detail = `HTTP ${res.status}`;
    54→    54→    try {
    55→    55→      const err = await res.json();
    56→    56→      detail = err.detail || detail;
    57→    57→    } catch { /* ignore */ }
    58→    58→    throw new Error(detail);
    59→    59→  }
    60→    60→  return res.json();
    61→    61→}

   128→   128→function handleLiveEvent(msg) {
   129→   129→  const { type, payload } = msg;
   130→   130→  if (type === 'initial_state' || type === 'world_sync' || type === 'scenario_restored') {
   131→   131→    applyWorldState(payload, false);
   132→   132→    if (type === 'scenario_restored') {
   133→   133→      sanctuary.scenarioDirector.clearStage();
   134→   134→      document.getElementById('scenario-director-bar').classList.add('hidden');
   135→   135→    }
   136→   136→  } else if (type === 'jarvis_interrupted') {
   137→   137→    applyWorldState(payload, false);
   138→   138→    sanctuary.jarvis.acknowledgeInterruption(sanctuary.camera.position);
   139→   139→  } else if (type === 'switch_changed') {
   140→   140→    applyWorldState(payload.world, false);
   141→   141→    enforceBrowserSensorGates();
   142→   142→  } else if (type === 'jarvis_spoke') {
   143→   143→    applyWorldState(payload.world, false);
   144→   144→    showJarvisResponse(payload.jarvis_message);
   145→   145→    if (payload.activated_scenario) {
   146→   146→      const def = state.scenarios.find((s) => s.id === payload.activated_scenario.scenario_id);
   147→   147→      if (def) activateClientScenario(def, payload.activated_scenario);
   148→   148→    }
   149→   149→  } else if (type === 'scenario_activated') {
   150→   150→    applyWorldState(payload.world, false);
   151→   151→    activateClientScenario(payload.scenario, payload.runtime);
   152→   152→  } else if (type === 'video_state_changed') {
   153→   153→    applyWorldState(payload.world, false);
   154→   154→    applyVideoState(payload.video);
   155→   155→  }
   156→   156→}

   272→   272→async function enforceBrowserSensorGates() {
   273→   273→  // Microphone enforcement
   274→   274→  if (!state.switches.microphone || !state.switches.master) {
   275→   275→    if (state.micStream) {
   276→   276→      state.micStream.getTracks().forEach((t) => t.stop());
   277→   277→      state.micStream = null;
   278→   278→    }
   279→   279→    if (state.recognition) {
   280→   280→      try { state.recognition.abort(); } catch { /* noop */ }
   281→   281→    }
   282→   282→  }
   283→   283→
   284→   284→  // Camera enforcement
   285→   285→  if (!state.switches.camera || !state.switches.master) {
   286→   286→    if (state.camStream) {
   287→   287→      state.camStream.getTracks().forEach((t) => t.stop());
   288→   288→      state.camStream = null;
   289→   289→      opticalCamEl.srcObject = null;
   290→   290→    }
   291→   291→    sanctuary.updateVisionBoard(false, null);
   292→   292→  } else if (!state.camStream && navigator.mediaDevices?.getUserMedia) {
   293→   293→    try {
   294→   294→      state.camStream = await navigator.mediaDevices.getUserMedia({ video: { width: 640, height: 360 } });
   295→   295→      opticalCamEl.srcObject = state.camStream;
   296→   296→      await opticalCamEl.play();
   297→   297→      sanctuary.updateVisionBoard(true, opticalCamEl);
   298→   298→    } catch {
   299→   299→      sanctuary.updateVisionBoard(false, null);
   300→   300→    }
   301→   301→  }
   302→   302→}
