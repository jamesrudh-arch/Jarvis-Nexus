    99→    99→async function api<T = Record<string, unknown>>(path: string, options: RequestInit = {}): Promise<T> {
   100→   100→  const headers = options.body instanceof FormData ? {} : { 'Content-Type': 'application/json' };
   101→   101→  const res = await fetch(path, { ...options, headers: { ...headers, ...(options.headers as Record<string, string> | undefined) } });
   102→   102→  if (!res.ok) {
   103→   103→    let detail = `HTTP ${res.status}`;
   104→   104→    try {
   105→   105→      const err = await res.json();
   106→   106→      detail = (err as { detail?: string }).detail || detail;
   107→   107→    } catch { /* ignore */ }
   108→   108→    throw new Error(detail);
   109→   109→  }
   110→   110→  return (await res.json()) as T;
   111→   111→}

   189→   189→  async bootstrap() {
   190→   190→    this.emit();
   191→   191→    // Yield one tick so React StrictMode's double-mount cleanup lands before
   192→   192→    // the heavy 14 MB character download starts (prevents a duplicate load).
   193→   193→    await new Promise((r) => setTimeout(r, 0));
   194→   194→    const sanctuary = this.sanctuary;
   195→   195→    if (!sanctuary || this.disposed) return;
   196→   196→
   197→   197→    try {
   198→   198→      await sanctuary.initCharacter((xhr: { loaded: number; total: number } | undefined) => {
   199→   199→        if (this.disposed) return;
   200→   200→        if (xhr && xhr.total) {
   201→   201→          const pct = Math.round((xhr.loaded / xhr.total) * 100);
   202→   202→          this.loadingText = `Loading rigged 3D character & skeletal clips (${pct}%)…`;
   203→   203→          this.emit();
   204→   204→        }
   205→   205→      });
   206→   206→    } catch (err) {
   207→   207→      this.loadingText = `Character asset warning: ${(err as Error).message}`;
   208→   208→      this.emit();
   209→   209→    }
   210→   210→    if (this.disposed) return;
   211→   211→
   212→   212→    try {
   213→   213→      const [worldSnap, scenariosRes] = await Promise.all([
   214→   214→        api<WorldSnapshot>('/api/v1/world'),
   215→   215→        api<{ scenarios: ScenarioDef[] }>('/api/v1/scenarios'),
   216→   216→      ]);
   217→   217→      if (this.disposed) return;
   218→   218→      this.scenarios = scenariosRes.scenarios || [];
   219→   219→      this.applyWorldState(worldSnap, true);
   220→   220→    } catch (err) {
   221→   221→      this.error = (err as Error).message;
   222→   222→      this.emit();
   223→   223→      return;
   224→   224→    }
   225→   225→
   226→   226→    this.ready = true;
   227→   227→    this.loadingText = 'NEXUS online.';
   228→   228→    this.emit();
   229→   229→
   230→   230→    // Animation loop
   231→   231→    sanctuary.renderer.setAnimationLoop(() => {
   232→   232→      const dt = this.animClock.getDelta();
   233→   233→      sanctuary.renderFrame(dt, this.opticalCamEl, Boolean(this.world?.switches.camera));
   234→   234→      this.tickScenarioUI();
   235→   235→    });
   236→   236→
   237→   237→    // Telemetry poll
   238→   238→    this.telemetryTimer = setInterval(async () => {
   239→   239→      try {
   240→   240→        this.telemetry = await api<Telemetry>('/api/v1/telemetry');
   241→   241→        sanctuary.updateTelemetryBoard({
   242→   242→          ...this.telemetry,
   243→   243→          inference: this.telemetry.inference,
   244→   244→        } as never);
   245→   245→        this.emit();
   246→   246→      } catch { /* transient */ }
   247→   247→    }, 8000);
   248→   248→    void this.fetchTelemetryOnce();
   249→   249→  }

   313→   313→      sanctuary.scenarioDirector.clearStage();
   314→   314→      this.activeScenarioDef = null;
   315→   315→    }
   316→   316→

   319→   319→    this.emit();
