    20→/** Scanline hologram shell material (fresnel edge boost + scanlines + flicker). */
    21→export class HoloScanMaterial extends THREE.ShaderMaterial {
    22→  constructor(colorHex, { opacity = 0.4, scanDensity = 22, fresnelPower = 2.4 } = {}) {
    23→    super({
    24→      uniforms: {
    25→        uColor: { value: new THREE.Color(colorHex) },
    26→        uTime: { value: 0 },
    27→        uOpacity: { value: opacity },
    28→        uScan: { value: scanDensity },
    29→        uFresnelPower: { value: fresnelPower },
    30→      },
    31→      vertexShader: `
    32→        varying vec3 vNormalW; varying vec3 vPosW; varying vec3 vPosL;
    33→        void main() {
    34→          vPosL = position;
    35→          vec4 wp = modelMatrix * vec4(position, 1.0);
    36→          vPosW = wp.xyz;
    37→          vNormalW = normalize(mat3(modelMatrix) * normal);
    38→          gl_Position = projectionMatrix * viewMatrix * wp;
    39→        }`,
    40→      fragmentShader: `
    41→        uniform vec3 uColor; uniform float uTime, uOpacity, uScan, uFresnelPower;
    42→        varying vec3 vNormalW; varying vec3 vPosW; varying vec3 vPosL;
    43→        void main() {
    44→          vec3 V = normalize(cameraPosition - vPosW);
    45→          float fres = pow(1.0 - abs(dot(normalize(vNormalW), V)), uFresnelPower);
    46→          float scan = 0.82 + 0.18 * sin(vPosL.y * uScan * 6.2831 - uTime * 2.4);
    47→          float flick = 0.93 + 0.07 * sin(uTime * 29.0) * sin(uTime * 17.3);
    48→          float a = (0.18 + 0.82 * fres) * scan * flick * uOpacity;
    49→          gl_FragColor = vec4(uColor * (0.5 + 0.95 * fres), a);
    50→        }`,
    51→      transparent: true,
    52→      depthWrite: false,
    53→      blending: THREE.AdditiveBlending,
    54→      side: THREE.DoubleSide,
    55→    });
    56→  }
    57→}

    59→/** Simple safe condition evaluator for branch rules ("key <= 1450" / "otherwise"). */
    60→function evalBranchCondition(cond, vars) {
    61→  if (!cond || cond === 'otherwise') return true;
    62→  const parts = cond.split('||');
    63→  for (const part of parts) {
    64→    const m = part.trim().match(/^(\w+)\s*(<=|>=|==|!=|<|>)\s*(-?[\d.]+)$/);
    65→    if (!m) continue;
    66→    const val = Number(vars[m[1]]);
    67→    const ref = Number(m[3]);
    68→    const op = m[2];
    69→    if (op === '<=') { if (val <= ref) return true; }
    70→    else if (op === '>=') { if (val >= ref) return true; }
    71→    else if (op === '==') { if (val === ref) return true; }
    72→    else if (op === '!=') { if (val !== ref) return true; }
    73→    else if (op === '<') { if (val < ref) return true; }
    74→    else if (op === '>') { if (val > ref) return true; }
    75→  }
    76→  return false;
    77→}

   193→  _sampleLiDAR(mesh, count, colorHex) {
   194→    try {
   195→      const sampler = new MeshSurfaceSampler(mesh).build();
   196→      const positions = new Float32Array(count * 3);
   197→      const colors = new Float32Array(count * 3);
   198→      const c = new THREE.Color(colorHex);
   199→      const cAlt = new THREE.Color(colorHex === CYAN ? ICE : AMBER);
   200→      const p = new THREE.Vector3();
   201→      for (let i = 0; i < count; i++) {
   202→        sampler.sample(p);
   203→        positions[i * 3] = p.x; positions[i * 3 + 1] = p.y; positions[i * 3 + 2] = p.z;
   204→        const cc = Math.random() > 0.92 ? cAlt : c;
   205→        colors[i * 3] = cc.r; colors[i * 3 + 1] = cc.g; colors[i * 3 + 2] = cc.b;
   206→      }
   207→      const geo = new THREE.BufferGeometry();
   208→      geo.setAttribute('position', new THREE.BufferAttribute(positions, 3));
   209→      geo.setAttribute('color', new THREE.BufferAttribute(colors, 3));
   210→      const pts = new THREE.Points(geo, new THREE.PointsMaterial({
   211→        size: 0.024, vertexColors: true, transparent: true, opacity: 0.85,
   212→        blending: THREE.AdditiveBlending, depthWrite: false,
   213→      }));

   857→      label: v.label,
   858→      value: this.variables[v.key] ?? v.default,
   859→      unit: v.unit || '',
   860→      tone: 'cyan',
   861→    }));
   862→
   863→    // Evaluate the active branch
   864→    let branch = null;
   865→    for (const b of def.branches || []) {
   866→      if (evalBranchCondition(b.condition, this.variables)) { branch = b; break; }
   867→    }
   868→
   869→    const series = this._computeSeries();
   870→
   871→    return {
   872→      title: def.title,
   873→      classification: def.classification,
   874→      phase,
   875→      t: this.currentTime,
   876→      duration: this.duration,
   877→      cues: tl.map((c) => ({ t: c.t, phase: c.phase })),
   878→      variables,
   879→      series,
   880→      branch,
   881→    };
   882→  }

   884→  _computeSeries() {
   885→    const kind = this.activeScenario?.actors_props?.voxel_structure;
   886→    const d = this.duration;
   887→    const pts = (fn, step = 0.5) => {
   888→      const out = [];
   889→      for (let t = 0; t <= d + 0.001; t += step) out.push({ t, v: fn(t) });
   890→      return out;
   891→    };
   892→    if (kind === 'cryo_manifold_riser') {
   893→      const pLine = Number(this.variables.line_pressure_mpa ?? 210);
   894→      const delayMs = Number(this.variables.relief_valve_ms ?? 125);
   895→      const mitigated = pLine < 185 || delayMs <= 70;
   896→      return [
   897→        {
   898→          name: 'Flange B-4 stress (MPa)', color: '#62e6f2', minV: 120, maxV: 260, unit: 'MPa',
   899→          points: pts((t) => (mitigated
   900→            ? 138 + Math.min(1, t / 11.2) * 14
   901→            : 138 + Math.pow(Math.max(0, (t - 4) / 7.2), 2.1) * (pLine - 138))),
   902→        },
   903→        {
   904→          name: 'Predicted unmitigated (inference)', color: '#e7ae5b', minV: 120, maxV: 260, unit: 'MPa',
   905→          points: pts((t) => 138 + Math.pow(Math.max(0, (t - 4) / 7.2), 2.1) * 82),
   906→        },
   907→      ];
   908→    }
   909→    if (kind === 'dual_turbines') {
   910→      const rpm = Number(this.variables.rotor_rpm ?? 11500);
   911→      const tempK = Number(this.variables.inlet_temp_k ?? 1320);
   912→      const stall = THREE.MathUtils.clamp((rpm - 10500) / 5500 + Math.max(0, tempK - 1450) / 900, 0, 1.4);
   913→      return [
   914→        { name: 'Variant A adiabatic eff. delta (%)', color: '#62e6f2', minV: 0, maxV: 8, unit: '%', points: pts((t) => 2.6 + Math.min(1, t / 5) * 0.5) },
   915→        { name: 'Variant B loss (tip vortex)', color: '#e7ae5b', minV: 0, maxV: 8, unit: '%', points: pts((t) => 1.1 + Math.min(1, t / 6) * stall * 4.2) },
   916→      ];
   917→    }
   918→    // zerotrust
   919→    const exposed = Number(this.variables.expose_public_port ?? 0) >= 1;
   920→    return [
   921→      { name: 'Authenticated outbound success (%)', color: '#62e6f2', minV: 0, maxV: 100, unit: '%', points: pts((t) => 96 + Math.sin(t * 1.4) * 3) },
   922→      { name: 'Unauthorized probe reach', color: exposed ? '#ff6b67' : '#e7ae5b', minV: 0, maxV: 100, unit: 'pct', points: pts((t) => (exposed ? Math.min(92, Math.max(0, (t - 2) * 18)) : 0)) },
   923→    ];
   924→  }

   929→  update(dt) {
   930→    if (!this.activeScenario || !this.stageRoot.visible) return;
   931→    this.clock += dt;
   932→    if (this.playing) {
   933→      this.currentTime = (this.currentTime + dt * this.playbackRate) % this.duration;
   934→    }
   935→
   936→    for (const m of this.holoMats) m.uniforms.uTime.value = this.clock;
   937→
   938→    const refs = {
   939→      t: this.currentTime,
   940→      duration: this.duration,
   941→      normT: this.currentTime / Math.max(1, this.duration),
   942→      that: this,
   943→    };
   944→    for (const fn of this.updatables) fn(dt, refs);
   945→
   946→    this._applyCueCamera();
   947→  }
