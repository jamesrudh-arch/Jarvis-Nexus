   597→   597→  renderFrame(dt, opticalVideoEl, cameraSwitchOn) {
   598→   598→    // 1. Update procedural video & volumetric idle elements
   599→   599→    this.schlierenFeed.update(dt);
   600→   600→    if (cameraSwitchOn && opticalVideoEl) {
   601→   601→      this.updateVisionBoard(true, opticalVideoEl);
   602→   602→    }
   603→   603→
   604→   604→    if (this.centerHoloAssembly) {

   606→   606→      const stageActive = Boolean(this.scenarioDirector.activeScenario);
   607→   607→      this.centerHoloAssembly.visible = !stageActive;
   608→   608→      if (!this.reducedMotion) {
   609→   609→        this.centerHoloAssembly.rotation.y += dt * 0.28;
   610→   610→      }
   611→   611→    }
   612→   612→
   613→   613→    if (!this.reducedMotion) {
   614→   614→      if (this.securityLattice) this.securityLattice.rotation.y += dt * 0.22;
   615→   615→      this.agentSolids.forEach((s, i) => {
   616→   616→        s.rotation.x += dt * (0.35 + i * 0.08);
   617→   617→        s.rotation.y += dt * 0.45;
   618→   618→      });
   619→   619→      if (this.extracted3DTwinGroup?.visible) {
   620→   620→        this.extracted3DTwinGroup.rotation.y += dt * 0.65;
   621→   621→      }
   622→   622→    }
   623→   623→
   624→   624→    // 2. Update Scenario Director, Living JARVIS, and Camera Director
   625→   625→    this.scenarioDirector.update(dt);
   626→   626→    this.jarvis.update(dt, this.camera.position, this.reducedMotion);
   627→   627→    this.cameraDirector.update(dt, this.jarvis, this.reducedMotion);
   628→   628→
   629→   629→    // 3. Project 3D world-space scenario callout tags onto screen
   630→   630→    this._renderSpatialAnchors();
   631→   631→
   632→   632→    // 4. Render either standard ACES Filmic sanctuary OR 4-Face Holographic Pyramid Scissor Viewports
   633→   633→    if (this.hologramMode && this.hologramLayout === 'pyramid_4face') {
   634→   634→      this._renderFourFacePyramid();
   635→   635→    } else {
   636→   636→      this.renderer.setScissorTest(false);
   637→   637→      this.composer.render();
   638→   638→    }

   641→   641→  _renderFourFacePyramid() {
   642→   642→    const w = window.innerWidth;
   643→   643→    const h = window.innerHeight;
   644→   644→    const size = Math.floor(Math.min(w, h) * 0.44);
   645→   645→    const cx = Math.floor(w / 2);
   646→   646→    const cy = Math.floor(h / 2);
   647→   647→
   648→   648→    this.renderer.setScissorTest(false);
   649→   649→    this.renderer.setClearColor(0x000000, 1);
   650→   650→    this.renderer.clear();
   651→   651→    this.renderer.setScissorTest(true);
   652→   652→
   653→   653→    const jPos = this.jarvis.root.position;
   654→   654→    const views = [
   655→   655→      { x: cx - size / 2, y: cy + 16, yaw: 0 },               // Top (0 deg)
   656→   656→      { x: cx + 16, y: cy - size / 2, yaw: Math.PI / 2 },     // Right (90 deg)
   657→   657→      { x: cx - size / 2, y: cy - size - 16, yaw: Math.PI },  // Bottom (180 deg)
   658→   658→      { x: cx - size - 16, y: cy - size / 2, yaw: -Math.PI / 2 }, // Left (270 deg)
   659→   659→    ];
   660→   660→
   661→   661→    for (const v of views) {
   662→   662→      const vx = Math.max(0, Math.floor(v.x));
   663→   663→      const vy = Math.max(0, Math.floor(v.y));
   664→   664→      const vw = Math.min(size, w - vx);
   665→   665→      const vh = Math.min(size, h - vy);
   666→   666→      if (vw <= 10 || vh <= 10) continue;
   667→   667→
   668→   668→      this.renderer.setViewport(vx, vy, vw, vh);
   669→   669→      this.renderer.setScissor(vx, vy, vw, vh);
   670→   670→      this.pyramidCam.aspect = vw / vh;
   671→   671→      this.pyramidCam.position.set(
   672→   672→        jPos.x + Math.sin(v.yaw) * 2.85,
   673→   673→        1.15,
   674→   674→        jPos.z + Math.cos(v.yaw) * 2.85
   675→   675→      );
   676→   676→      this.pyramidCam.lookAt(jPos.x, 0.98, jPos.z);
   677→   677→      this.pyramidCam.updateProjectionMatrix();
   678→   678→      this.renderer.render(this.scene, this.pyramidCam);
   679→   679→    }
   680→   680→    this.renderer.setViewport(0, 0, w, h);
   681→   681→    this.renderer.setScissorTest(false);
   682→   682→  }
