   112→  async load(onProgress) {
   113→    const gltfLoader = new GLTFLoader();
   114→    const fbxLoader = new FBXLoader();
   115→
   116→    const [avatarGltf, xbotGltf, walkFbx] = await Promise.all([
   117→      gltfLoader.loadAsync('/models/jarvis_avaturn.glb', onProgress),
   118→      gltfLoader.loadAsync('/models/xbot_locomotion.glb'),
   119→      fbxLoader.loadAsync('/models/walking.fbx'),
   120→    ]);
   121→
   122→    this.armature = avatarGltf.scene.getObjectByName('Armature') || avatarGltf.scene;
   123→    this.armature.scale.setScalar(1);
   124→
   125→    // Calibrate skeleton to standard Mixamo coordinate rolls & scale
   126→    retarget(this.armature, AVATURN_RETARGET);
   127→
   128→    // --- Cross-rig quaternion retargeting tables -------------------------------
   129→    // The Avaturn rig's leg bones rest ~180deg (Z) away from the Mixamo
   130→    // convention, so raw Mixamo clip quaternions would flip the legs upward.
   131→    // Every clip quaternion is therefore converted delta-from-rest:
   132→    //   q_target = q_targetRest * inverse(q_sourceRest) * q_sourceAnim
   133→    const captureRest = (root) => {
   134→      const rest = {};
   135→      root.traverse((o) => {
   136→        if (o.isBone) {
   137→          const clean = o.name.replace(/^mixamorig:/, '').replace(/^mixamorig/, '');
   138→          if (!rest[clean]) rest[clean] = o.quaternion.clone();
   139→        }
   140→      });
   141→      return rest;
   142→    };
   143→    const targetRest = captureRest(this.armature); // post-retarget bind pose
   144→    const xbotRest = captureRest(xbotGltf.scene);
   145→    const walkRest = captureRest(walkFbx);
   146→    this._retargetTables = { targetRest, xbotRest, walkRest };
   147→
   148→    // Tailor PBR materials to match approved Storyboard Character Direction
   149→    this.armature.traverse((obj) => {
   150→      if (obj.isBone) {
   151→        this.bones[obj.name] = obj;
   152→      }
   153→      if (obj.isMesh) {
   154→        obj.castShadow = true;
   155→        obj.receiveShadow = true;
   156→        obj.frustumCulled = false;
   157→
   158→        if (obj.morphTargetDictionary && obj.morphTargetInfluences) {
   159→          this.morphMeshes.push(obj);
   160→        }
   161→
   162→        const mat = obj.material;
   163→        if (mat) {
   164→          const mName = (mat.name || obj.name || '').toLowerCase();

   168→            mat.metalness = 0.06;

   171→            mat.color = new THREE.Color(0x2a1d17);
   172→            mat.roughness = 0.64;
   173→            mat.metalness = 0.04;

   177→            mat.roughness = 0.42;
   178→            mat.metalness = 0.12;
   179→            mat.needsUpdate = true;
   180→          } else if (mName === 'head' || mName === 'body') {
   181→            const phys = new THREE.MeshPhysicalMaterial({
   182→              map: mat.map,
   183→              normalMap: mat.normalMap,
   184→              roughnessMap: mat.roughnessMap,

   186→              metalness: 0.0,

   188→              sheenRoughness: 0.68,

   214→        .setFromEuler(new THREE.Euler(0.2, 0, -0.62))

   234→  _prepareMixamoClip(clip, newName, posScale = 0.01, stripXZRoot = true, srcRest = null, tgtRest = null, armCarry = null) {
   235→    clip.name = newName;
   236→    const keptTracks = [];
   237→    const tmpQ = new THREE.Quaternion();
   238→    for (const t of clip.tracks) {
   239→      t.name = t.name.replaceAll('mixamorig:', '').replaceAll('mixamorig', '');
   240→      const [boneName, prop] = t.name.split('.');
   241→      if (!this.bones[boneName]) continue;
   242→
   243→      if (prop === 'quaternion') {
   244→        // Delta-from-rest conversion between source rig and Avaturn bind pose
   245→        const qa = srcRest ? srcRest[boneName] : null;
   246→        const qb = tgtRest ? tgtRest[boneName] : null;
   247→        if (qa && qb) {
   248→          const qaInv = qa.clone().invert();
   249→          for (let i = 0; i < t.values.length; i += 4) {
   250→            tmpQ.set(t.values[i], t.values[i + 1], t.values[i + 2], t.values[i + 3]);
   251→            tmpQ.premultiply(qaInv).premultiply(qb);
   252→            t.values[i] = tmpQ.x;
   253→            t.values[i + 1] = tmpQ.y;
   254→            t.values[i + 2] = tmpQ.z;
   255→            t.values[i + 3] = tmpQ.w;
   256→          }
   257→        }
   258→        // Optional natural arm carry (post-multiplied local rotation)
   259→        if (armCarry) {
   260→          const carry = boneName === 'LeftArm' ? armCarry.left : (boneName === 'RightArm' ? armCarry.right : null);
   261→          if (carry) {
   262→            for (let i = 0; i < t.values.length; i += 4) {
   263→              tmpQ.set(t.values[i], t.values[i + 1], t.values[i + 2], t.values[i + 3]);
   264→              tmpQ.multiply(carry);
   265→              t.values[i] = tmpQ.x;
   266→              t.values[i + 1] = tmpQ.y;
   267→              t.values[i + 2] = tmpQ.z;
   268→              t.values[i + 3] = tmpQ.w;
   269→            }
   270→          }
   271→        }
   272→        keptTracks.push(t);
   273→        continue;
   274→      }
   275→
   276→      if (prop === 'position') {
   277→        if (boneName !== 'Hips') continue;
   278→        const baseY = t.values[1] * (t.values[1] > 3 ? posScale : 1.0);
   279→        const factor = t.values[1] > 3 ? posScale : 1.0;
   280→        for (let i = 0; i < t.values.length; i += 3) {
   281→          t.values[i] = stripXZRoot ? 0 : t.values[i] * factor;
   282→          t.values[i + 1] = Math.max(0.92, t.values[i + 1] * factor || baseY);
   283→          t.values[i + 2] = stripXZRoot ? 0 : t.values[i + 2] * factor;
   284→        }
   285→      }
   286→      keptTracks.push(t);
   287→    }
   288→    clip.tracks = keptTracks;
   289→    return clip;
   290→  }

   292→  _fadeToAction(name, duration = 0.32) {
   293→    if (!this.actions[name] || this.activeActionName === name) return;
   294→    const next = this.actions[name];
   295→    next.reset().setEffectiveTimeScale(1).setEffectiveWeight(1).fadeIn(duration).play();
   296→    if (this.activeActionName && this.actions[this.activeActionName]) {
   297→      this.actions[this.activeActionName].fadeOut(duration);
   298→    }
   299→    this.activeActionName = name;
   300→  }

   302→  /**
   303→   * Navigate smoothly around the central holo-table obstacle (radius 2.15m at origin)
   304→   * so JARVIS never clips through consoles or slides sideways.
   305→   */
   306→  navigateTo(targetX, targetZ, targetYaw, { urgent = false, posture = 'console_interaction', emotion = 'focused' } = {}) {
   307→    this.urgentRun = Boolean(urgent);
   308→    this.postureMode = posture;
   309→    this.emotionalState = emotion;
   310→    this.targetYaw = targetYaw;
   311→
   312→    const start = new THREE.Vector2(this.root.position.x, this.root.position.z);
   313→    const goal = new THREE.Vector2(targetX, targetZ);
   314→    const dist = start.distanceTo(goal);
   315→    if (dist < 0.14) {
   316→      this.waypoints = [];
   317→      return;
   318→    }
   319→
   320→    // Check if straight segment intersects central circular holo-table (r = 1.95m)
   321→    const seg = goal.clone().sub(start);
   322→    const segLenSq = seg.lengthSq();
   323→    const tProj = Math.max(0, Math.min(1, (-start.x * seg.x - start.z * seg.y) / segLenSq));
   324→    const closest = start.clone().addScaledVector(seg, tProj);
   325→    const clearanceRadius = 2.15;
   326→
   327→    this.waypoints = [];
   328→    if (closest.length() < clearanceRadius && start.length() > 1.2 && goal.length() > 1.2) {
   329→      // Compute tangent arc waypoint around the central ring
   330→      const a1 = Math.atan2(start.y, start.x);
   331→      const a2 = Math.atan2(goal.y, goal.x);
   332→      let delta = a2 - a1;
   333→      while (delta > Math.PI) delta -= Math.PI * 2;
   334→      while (delta < -Math.PI) delta += Math.PI * 2;
   335→      const midAng = a1 + delta * 0.5;
   336→      this.waypoints.push(new THREE.Vector2(Math.cos(midAng) * 2.55, Math.sin(midAng) * 2.55));
   337→    }
   338→    this.waypoints.push(goal);
   339→  }

   341→  acknowledgeInterruption(cameraWorldPos) {
   342→    this.waypoints = [];
   343→    this.currentSpeed = 0;
   344→    this.urgentRun = false;
   345→    this.postureMode = 'conversational_focus';

   355→  /**
   356→   * Speak with real-time phoneme-to-Oculus-viseme facial synchronization.
   357→   * Primary voice: server-synthesized audio (base64) with the viseme timeline
   358→   * elastically scaled to the real audio duration. Fallback: browser SpeechSynthesis.
   359→   */
   360→  speak(text, { onStart, onEnd, audio } = {}) {
   361→    this.interruptSpeech();
   362→    if (!text) return;
   363→
   364→    const clean = text.replace(/\[.*?\]/g, '').trim();
   365→    if (!clean) return;
   366→
   367→    // Build phoneme/viseme schedule using LipsyncEn
   368→    const words = clean.split(/\s+/);
   369→    const timeline = [];
   370→    let cursorMs = 60;
   371→
   372→    for (const rawWord of words) {
   373→      const upper = this.lipsync.preProcessText(rawWord);
   374→      const oculusSeq = this.lipsync.wordsToVisemes(upper);
   375→      const wordDurMs = Math.max(150, Math.min(540, rawWord.length * 58));
   376→
   377→      if (oculusSeq && oculusSeq.visemes && oculusSeq.visemes.length > 0) {
   378→        const totalRel = oculusSeq.durations.reduce((a, b) => a + b, 0) || 1;
   379→        for (let i = 0; i < oculusSeq.visemes.length; i++) {
   380→          const vCode = oculusSeq.visemes[i];
   381→          const segMs = (oculusSeq.durations[i] / totalRel) * wordDurMs;
   382→          timeline.push({
   383→            viseme: `viseme_${vCode}`,
   384→            startMs: cursorMs,
   385→            endMs: cursorMs + segMs,
   386→          });
   387→          cursorMs += segMs;
   388→        }
   389→      } else {
   390→        timeline.push({ viseme: 'viseme_aa', startMs: cursorMs, endMs: cursorMs + wordDurMs * 0.7 });
   391→        cursorMs += wordDurMs;
   392→      }
   393→      cursorMs += 42;
   394→    }
   395→
   396→    this.speechTimeline = timeline;
   397→    this.speechStartPerf = performance.now();
   398→    this.isSpeaking = true;
   399→    this.onSpeechEndCallback = onEnd || null;
   400→    if (onStart) onStart();
   401→
   402→    // Primary: server-synthesized NEXUS voice (base64 WAV) with elastic viseme scaling
   403→    if (audio && audio.data) {
   404→      try {
   405→        const snd = new Audio(`data:${audio.mime || 'audio/wav'};base64,${audio.data}`);
   406→        this.activeAudioEl = snd;
   407→        const lastEndMs = timeline.length ? timeline[timeline.length - 1].endMs : 1;
   408→        snd.onloadedmetadata = () => {
   409→          const durMs = (snd.duration || 0) * 1000;
   410→          if (durMs > 400 && lastEndMs > 0) {
   411→            const factor = durMs / lastEndMs;
   412→            for (const seg of this.speechTimeline) {
   413→              seg.startMs *= factor;
   414→              seg.endMs *= factor;
   415→            }
   416→          }
   417→        };
   418→        snd.onended = () => {
   419→          this.activeAudioEl = null;
   420→          this.isSpeaking = false;
   421→          this._clearVisemes();
   422→          if (this.onSpeechEndCallback) {
   423→            const cb = this.onSpeechEndCallback;
   424→            this.onSpeechEndCallback = null;
   425→            cb();
   426→          }
   427→        };
   428→        snd.play().catch(() => this._browserTtsFallback(clean));
   429→        return;
   430→      } catch {
   431→        this.activeAudioEl = null;
   432→      }
   433→    }
   434→
   435→    this._browserTtsFallback(clean);
   436→  }
   437→
   438→  _browserTtsFallback(clean) {
   439→    // Synthesize audible voice via browser SpeechSynthesis when server audio is unavailable
   440→    if ('speechSynthesis' in window) {
   441→      try {
   442→        window.speechSynthesis.cancel();
   443→        const utt = new SpeechSynthesisUtterance(clean);
   444→        const voices = window.speechSynthesis.getVoices();
   445→        const femaleVoice = voices.find((v) =>
   446→          /en/i.test(v.lang) && /samantha|victoria|kate|serena|sonia|aria|jenny|female|uk/i.test(v.name)
   447→        ) || voices.find((v) => /en-GB|en-US/i.test(v.lang));
   448→        if (femaleVoice) utt.voice = femaleVoice;
   449→        utt.rate = 1.02;
   450→        utt.pitch = 0.96;
   451→        utt.onend = () => {
   452→          this.isSpeaking = false;
   453→          this._clearVisemes();
   454→          if (this.onSpeechEndCallback) this.onSpeechEndCallback();
   455→        };
   456→        this.activeUtterance = utt;
   457→        window.speechSynthesis.speak(utt);
   458→      } catch {
   459→        // Fallback purely to timed viseme timeline
   460→      }
   461→    }
   462→  }
   463→
   464→  interruptSpeech() {
   465→    if (this.activeAudioEl) {
   466→      try { this.activeAudioEl.pause(); } catch { /* noop */ }
   467→      this.activeAudioEl = null;
   468→    }
   469→    if ('speechSynthesis' in window) {
   470→      try { window.speechSynthesis.cancel(); } catch { /* noop */ }
   471→    }
   472→    this.isSpeaking = false;
   473→    this.speechTimeline = [];
   474→    this.activeUtterance = null;
   475→    this._clearVisemes();
   476→    if (this.onSpeechEndCallback) {
   477→      const cb = this.onSpeechEndCallback;
   478→      this.onSpeechEndCallback = null;
   479→      cb();
   480→    }
   481→  }

   483→  _setMorph(name, value, lerpAlpha = 0.25) {
   484→    for (const mesh of this.morphMeshes) {
   485→      const idx = mesh.morphTargetDictionary[name];
   486→      if (idx !== undefined) {
   487→        const cur = mesh.morphTargetInfluences[idx] || 0;
   488→        mesh.morphTargetInfluences[idx] = cur + (value - cur) * lerpAlpha;
   489→      }
   490→    }
   491→  }

   493→  _clearVisemes() {
   494→    for (const v of VISEME_NAMES) {
   495→      this._setMorph(v, 0, 1.0);
   496→    }
   497→    this._setMorph('jawOpen', 0, 1.0);
   498→  }
   499→
   500→  getHeadWorldPosition(target = new THREE.Vector3()) {
   501→    if (this.bones.Head) {
   502→      this.bones.Head.getWorldPosition(target);
   503→      return target;
   504→    }
   505→    return target.set(this.root.position.x, 1.62, this.root.position.z);
   506→  }
   507→
   508→  update(dt, cameraWorldPos, reducedMotion = false) {
   509→    if (!this.loaded) return;
   510→    const clampedDt = Math.min(dt, 0.05);
   511→    this.breathClock += clampedDt;
   512→
   513→    // 1. Grounded Waypoint Locomotion (zero foot-skating!)
   514→    if (this.waypoints.length > 0) {
   515→      const wp = this.waypoints[0];
   516→      const dx = wp.x - this.root.position.x;
   517→      const dz = wp.y - this.root.position.z;
   518→      const dist = Math.hypot(dx, dz);
   519→
   520→      if (dist < 0.12) {
   521→        this.waypoints.shift();
   522→      } else {
   523→        const desiredYaw = Math.atan2(dx, dz);
   524→        this.root.rotation.y = this._lerpAngle(this.root.rotation.y, desiredYaw, clampedDt * 7.5);
   525→
   526→        const maxSpeed = this.urgentRun ? 3.25 : 1.32;
   527→        const decelFactor = this.waypoints.length === 1 ? Math.min(1, dist / 0.75) : 1;
   528→        const targetSpeed = maxSpeed * Math.max(0.32, decelFactor);
   529→        this.currentSpeed += (targetSpeed - this.currentSpeed) * Math.min(1, clampedDt * 8.0);
   530→
   531→        const step = Math.min(dist, this.currentSpeed * clampedDt);
   532→        this.root.position.x += (dx / dist) * step;
   533→        this.root.position.z += (dz / dist) * step;
   534→
   535→        const clipName = this.urgentRun && this.actions.run ? 'run' : 'walk';
   536→        this._fadeToAction(clipName, 0.22);
   537→        if (this.actions[clipName]) {
   538→          const refSpeed = clipName === 'run' ? 3.25 : 1.32;
   539→          this.actions[clipName].setEffectiveTimeScale(Math.max(0.55, this.currentSpeed / refSpeed));
   540→        }
   541→      }
   542→    } else {
   543→      this.currentSpeed *= Math.max(0, 1 - clampedDt * 10);
   544→      this._fadeToAction('idle', 0.35);
   545→      this.root.rotation.y = this._lerpAngle(this.root.rotation.y, this.targetYaw, clampedDt * 5.0);
   546→    }
   547→
   548→    // 2. Advance skeletal mixer
   549→    this.mixer.update(clampedDt);
   550→
   551→    // 3. Subtle additive posture & hand interaction layering (only when standing still)
   552→    if (this.waypoints.length === 0 && !reducedMotion) {
   553→      this._applyPostureLayer();
   554→    }
   555→
   556→    // 4. Head & Eye Gaze Tracking (rotates eyes/head naturally rather than spinning whole body!)
   557→    const gazeTarget = this.focusCamera ? cameraWorldPos : this.lookWorldTarget;
   558→    this._applyGazeTracking(gazeTarget);
   559→
   560→    // 5. Asymmetric Natural Blinking
   561→    this._updateBlinking(clampedDt);
   562→
   563→    // 6. Emotional Facial Expression Morphs
   564→    this._updateExpressions();
   565→
   566→    // 7. Synchronized Phoneme/Viseme Lip-Sync
   567→    this._updateLipSync();
   568→  }
   569→
   570→  _applyPostureLayer() {
   571→    const t = this.breathClock;
   572→    if (this.bones.Spine1) {
   573→      this.bones.Spine1.rotation.x += Math.sin(t * 1.45) * 0.012;
   574→    }
   575→
   576→    const rArm = this.bones.RightArm;
   577→    const rFore = this.bones.RightForeArm;
   578→    const lArm = this.bones.LeftArm;
   579→    const lFore = this.bones.LeftForeArm;
   580→    if (!rArm || !rFore || !lArm || !lFore) return;
   581→
   582→    if (this.postureMode === 'console_interaction') {
   583→      // Right hand raised gently manipulating volumetric assembly above holo-table
   584→      rArm.rotation.x -= 0.42 + Math.sin(t * 1.1) * 0.04;
   585→      rArm.rotation.z -= 0.12;
   586→      rFore.rotation.x -= 0.55 + Math.cos(t * 1.3) * 0.05;
   587→    } else if (this.postureMode === 'scenario_construct') {
   588→      // Symmetrical restrained two-hand volumetric expansion gesture
   589→      rArm.rotation.x -= 0.48;
   590→      rFore.rotation.x -= 0.52;
   591→      lArm.rotation.x -= 0.48;
   592→      lFore.rotation.x -= 0.52;
   593→    } else if (this.postureMode === 'scenario_direct' || this.postureMode === 'video_present') {
   594→      //Measured explanatory presentation gesture toward 3D evidence
   595→      rArm.rotation.x -= 0.54;
   596→      rArm.rotation.y += 0.22;
   597→      rFore.rotation.x -= 0.38;
   598→    } else if (this.isSpeaking) {
   599→      // Restrained conversational gesture (not on every word!)
   600→      const env = Math.max(0, Math.sin(t * 0.85));
   601→      rArm.rotation.x -= 0.22 * env;
   602→      rFore.rotation.x -= 0.38 * env;
   603→    }
   604→  }
   605→
   606→  _applyGazeTracking(worldTarget) {
   607→    if (!this.bones.Head || !this.bones.Neck) return;
   608→    const headPos = this.getHeadWorldPosition(new THREE.Vector3());
   609→    const localDir = worldTarget.clone().sub(headPos);
   610→    localDir.applyAxisAngle(new THREE.Vector3(0, 1, 0), -this.root.rotation.y).normalize();
   611→
   612→    const yaw = THREE.MathUtils.clamp(Math.atan2(localDir.x, localDir.z), -0.58, 0.58);
   613→    const pitch = THREE.MathUtils.clamp(-Math.asin(THREE.MathUtils.clamp(localDir.y, -0.8, 0.8)), -0.32, 0.32);
   614→
   615→    this.bones.Neck.rotation.y += yaw * 0.32;
   616→    this.bones.Neck.rotation.x += pitch * 0.28;
   617→    this.bones.Head.rotation.y += yaw * 0.45;
   618→    this.bones.Head.rotation.x += pitch * 0.42;
   619→
   620→    // Subtle ocular look morphs
   621→    const eyeX = THREE.MathUtils.clamp(yaw * 0.85, -0.5, 0.5);
   622→    const eyeY = THREE.MathUtils.clamp(-pitch * 0.85, -0.4, 0.4);
   623→    this._setMorph('eyeLookOutLeft', Math.max(0, eyeX), 0.3);
   624→    this._setMorph('eyeLookInRight', Math.max(0, eyeX), 0.3);
   625→    this._setMorph('eyeLookInLeft', Math.max(0, -eyeX), 0.3);
   626→    this._setMorph('eyeLookOutRight', Math.max(0, -eyeX), 0.3);
   627→    this._setMorph('eyeLookUpLeft', Math.max(0, eyeY), 0.3);
   628→    this._setMorph('eyeLookUpRight', Math.max(0, eyeY), 0.3);
   629→    this._setMorph('eyeLookDownLeft', Math.max(0, -eyeY), 0.3);
   630→    this._setMorph('eyeLookDownRight', Math.max(0, -eyeY), 0.3);
   631→  }
   632→
   633→  _updateBlinking(dt) {
   634→    this.blinkTimer -= dt;
   635→    if (this.blinkTimer <= 0 && this.blinkPhase < 0) {
   636→      this.blinkPhase = 0;
   637→      this.blinkTimer = 3.1 + Math.random() * 2.8;
   638→    }
   639→    if (this.blinkPhase >= 0) {
   640→      this.blinkPhase += dt / 0.15;
   641→      const w = this.blinkPhase < 0.5 ? this.blinkPhase * 2 : Math.max(0, (1 - this.blinkPhase) * 2);
   642→      this._setMorph('eyeBlinkLeft', w, 0.85);
   643→      this._setMorph('eyeBlinkRight', w * 0.96, 0.85);
   644→      if (this.blinkPhase >= 1) {
   645→        this.blinkPhase = -1;
   646→        this._setMorph('eyeBlinkLeft', 0, 0.85);
   647→        this._setMorph('eyeBlinkRight', 0, 0.85);
   648→      }
   649→    }
   650→  }
   651→
   652→  _updateExpressions() {
   653→    const e = this.emotionalState;
   654→    this._setMorph('browInnerUp', e === 'listening' ? 0.18 : (e === 'concerned' ? 0.36 : (e === 'urgent' ? 0.24 : 0.04)), 0.14);
   655→    this._setMorph('browDownLeft', e === 'focused' ? 0.24 : (e === 'concerned' ? 0.22 : (e === 'urgent' ? 0.3 : 0.0)), 0.14);
   656→    this._setMorph('browDownRight', e === 'focused' ? 0.24 : (e === 'concerned' ? 0.22 : (e === 'urgent' ? 0.3 : 0.0)), 0.14);
   657→    this._setMorph('eyeSquintLeft', e === 'focused' ? 0.18 : 0.04, 0.14);
   658→    this._setMorph('eyeSquintRight', e === 'focused' ? 0.18 : 0.04, 0.14);
   659→    this._setMorph('eyeWideLeft', e === 'urgent' ? 0.22 : 0.0, 0.18);
   660→    this._setMorph('eyeWideRight', e === 'urgent' ? 0.22 : 0.0, 0.18);
   661→    // Composed poise — never a frozen customer-support grin
   662→    this._setMorph('mouthSmileLeft', e === 'listening' ? 0.06 : 0.02, 0.12);
   663→    this._setMorph('mouthSmileRight', e === 'listening' ? 0.06 : 0.02, 0.12);
   664→  }
   665→
   666→  _updateLipSync() {
   667→    if (!this.isSpeaking || this.speechTimeline.length === 0) {
   668→      for (const v of VISEME_NAMES) this._setMorph(v, 0, 0.25);
   669→      this._setMorph('jawOpen', 0, 0.25);
   670→      return;
   671→    }
   672→
   673→    const elapsed = performance.now() - this.speechStartPerf;
   674→    const lastEnd = this.speechTimeline[this.speechTimeline.length - 1].endMs;
   675→
   676→    if (elapsed > lastEnd + 180) {
   677→      const audioStillPlaying = this.activeAudioEl && !this.activeAudioEl.paused;
   678→      if (!audioStillPlaying && (!('speechSynthesis' in window) || !window.speechSynthesis.speaking)) {
   679→        this.isSpeaking = false;
   680→        this._clearVisemes();
   681→        if (this.onSpeechEndCallback) {
   682→          const cb = this.onSpeechEndCallback;
   683→          this.onSpeechEndCallback = null;
   684→          cb();
   685→        }
   686→      }
   687→      return;
   688→    }
   689→
   690→    let activeV = 'viseme_sil';
   691→    for (const seg of this.speechTimeline) {
   692→      if (elapsed >= seg.startMs && elapsed <= seg.endMs) {
   693→        activeV = seg.viseme;
   694→        break;
   695→      }
   696→    }
   697→
   698→    for (const v of VISEME_NAMES) {
   699→      this._setMorph(v, v === activeV ? 0.68 : 0.0, 0.42);
   700→    }
   701→    const openWeight = ['viseme_aa', 'viseme_O', 'viseme_E'].includes(activeV) ? 0.26 : 0.08;
   702→    this._setMorph('jawOpen', activeV === 'viseme_sil' ? 0.0 : openWeight, 0.38);
   703→  }
   704→
   705→  _lerpAngle(a, b, t) {
   706→    let diff = b - a;
   707→    while (diff > Math.PI) diff -= Math.PI * 2;
   708→    while (diff < -Math.PI) diff += Math.PI * 2;
   709→    return a + diff * Math.min(1, t);
   710→  }
