   371→  /**
   372→   * Camera-facing catchlight sprites on both eyes — the single cheapest trick that
   373→   * turns "painted dead eyes" into a living gaze. Parented to the head bone with
   374→   * offsets calibrated against the Avaturn rig; modulated during blinks.
   375→   */
   376→  _buildEyeCatchlights() {
   377→    const c = document.createElement('canvas');
   378→    c.width = c.height = 64;
   379→    const ctx = c.getContext('2d');
   380→    // Hard corneal glint core + faint ambient halo — the two-part catchlight
   381→    // that real wet eyes have (a hard key glint + a soft broad reflection).
   382→    const halo = ctx.createRadialGradient(32, 32, 2, 32, 32, 30);
   383→    halo.addColorStop(0, 'rgba(235,250,255,0.34)');
   384→    halo.addColorStop(0.4, 'rgba(190,235,248,0.1)');
   385→    halo.addColorStop(1, 'rgba(160,225,240,0)');
   386→    ctx.fillStyle = halo;
   387→    ctx.fillRect(0, 0, 64, 64);
   388→    ctx.fillStyle = 'rgba(255,255,255,1)';
   389→    ctx.beginPath();
   390→    ctx.arc(25, 25, 8.5, 0, Math.PI * 2);
   391→    ctx.fill();
   392→    ctx.fillStyle = 'rgba(245,253,255,0.92)';
   393→    ctx.beginPath();
   394→    ctx.arc(40, 37, 3.6, 0, Math.PI * 2);
   395→    ctx.fill();
   396→    const tex = new THREE.CanvasTexture(c);
   397→
   398→    const head = this.bones.Head;
   399→    if (!head) return;
   400→    this.catchlights = [];
   401→    for (const sideSign of [-1, 1]) {
   402→      const mat = new THREE.SpriteMaterial({
   403→        map: tex,
   404→        transparent: true,
   405→        opacity: sideSign < 0 ? 1.0 : 0.82, // asymmetric — the far eye catches less key
   406→        depthWrite: false,
   407→      });
   408→      const sp = new THREE.Sprite(mat);
   409→      sp.scale.setScalar(sideSign < 0 ? 0.038 : 0.03);
   410→      sp.position.set(sideSign * 0.031, 0.052, 0.093);
   411→      head.add(sp);
   412→      this.catchlights.push(sp);
   413→      this._catchlightBase = this._catchlightBase || [];
   414→      this._catchlightBase.push(sp.scale.x);
   415→
   416→      // Lower-lid tear-film glint — a thin wet streak under each eye. The
   417→      // classic VFX tell of moist, living eyes.
   418→      const wetMat = new THREE.SpriteMaterial({
   419→        map: tex,
   420→        transparent: true,
   421→        opacity: sideSign < 0 ? 0.55 : 0.4,
   422→        depthWrite: false,
   423→      });
   424→      const wet = new THREE.Sprite(wetMat);
   425→      wet.scale.set(sideSign < 0 ? 0.026 : 0.02, 0.0065, 1);
   426→      wet.position.set(sideSign * 0.031, 0.036, 0.093);
   427→      head.add(wet);
   428→    }
   429→  }

   453→  /**
   454→   * Reposition the 3-point rig around her head relative to the ACTIVE camera:

   459→  _updateCameraRelativeLighting(cameraWorldPos) {
   460→    if (!this.keyLight) return;
   461→    const head = this.getHeadWorldPosition(this._tmpHead || (this._tmpHead = new THREE.Vector3()));
   462→    const toCam = this._tmpDir || (this._tmpDir = new THREE.Vector3());
   463→    toCam.copy(cameraWorldPos).sub(head);
   464→    toCam.y = Math.max(toCam.y, 0.15); // keep the geometry stable when camera dips
   465→    toCam.normalize();
   466→    const side = this._tmpSide || (this._tmpSide = new THREE.Vector3());
   467→    side.crossVectors(toCam, THREE.Object3D.DEFAULT_UP);
   468→    if (side.lengthSq() < 1e-4) side.set(1, 0, 0);
   469→    side.normalize();
   470→

   516→  _prepareMixamoClip(clip, newName, posScale = 0.01, stripXZRoot = true, srcRest = null, tgtRest = null, armCarry = null) {
   517→    clip.name = newName;
   518→    const keptTracks = [];
   519→    const tmpQ = new THREE.Quaternion();
   520→    for (const t of clip.tracks) {
   521→      t.name = t.name.replaceAll('mixamorig:', '').replaceAll('mixamorig', '');
   522→      const [boneName, prop] = t.name.split('.');
   523→      if (!this.bones[boneName]) continue;
   524→
   525→      if (prop === 'quaternion') {
   526→        // Delta-from-rest conversion between source rig and Avaturn bind pose
   527→        const qa = srcRest ? srcRest[boneName] : null;
   528→        const qb = tgtRest ? tgtRest[boneName] : null;
   529→        if (qa && qb) {
   530→          const qaInv = qa.clone().invert();
   531→          for (let i = 0; i < t.values.length; i += 4) {
   532→            tmpQ.set(t.values[i], t.values[i + 1], t.values[i + 2], t.values[i + 3]);
   533→            tmpQ.premultiply(qaInv).premultiply(qb);
   534→            t.values[i] = tmpQ.x;
   535→            t.values[i + 1] = tmpQ.y;
   536→            t.values[i + 2] = tmpQ.z;
   537→            t.values[i + 3] = tmpQ.w;
   538→          }
   539→        }
   540→        // Optional natural arm carry (post-multiplied local rotation)
   541→        if (armCarry) {
   542→          const carry = boneName === 'LeftArm' ? armCarry.left : (boneName === 'RightArm' ? armCarry.right : null);
   543→          if (carry) {
   544→            for (let i = 0; i < t.values.length; i += 4) {
   545→              tmpQ.set(t.values[i], t.values[i + 1], t.values[i + 2], t.values[i + 3]);
   546→              tmpQ.multiply(carry);
   547→              t.values[i] = tmpQ.x;
   548→              t.values[i + 1] = tmpQ.y;
   549→              t.values[i + 2] = tmpQ.z;
   550→              t.values[i + 3] = tmpQ.w;
   551→            }
   552→          }
   553→        }
   554→        keptTracks.push(t);
   555→        continue;
   556→      }
   557→
   558→      if (prop === 'position') {
   559→        if (boneName !== 'Hips') continue;
   560→        const baseY = t.values[1] * (t.values[1] > 3 ? posScale : 1.0);
   561→        const factor = t.values[1] > 3 ? posScale : 1.0;
   562→        for (let i = 0; i < t.values.length; i += 3) {
   563→          t.values[i] = stripXZRoot ? 0 : t.values[i] * factor;
   564→          t.values[i + 1] = Math.max(0.92, t.values[i + 1] * factor || baseY);
   565→          t.values[i + 2] = stripXZRoot ? 0 : t.values[i + 2] * factor;
   566→        }
   567→      }
   568→      keptTracks.push(t);
   569→    }
   570→    clip.tracks = keptTracks;
   571→    return clip;
   572→  }

   584→  /**
   585→   * Navigate smoothly around the central holo-table obstacle (radius 2.15m at origin)
   586→   * so JARVIS never clips through consoles or slides sideways.
   587→   */
   588→  navigateTo(targetX, targetZ, targetYaw, { urgent = false, posture = 'console_interaction', emotion = 'focused' } = {}) {
   589→    this.urgentRun = Boolean(urgent);
   590→    this.postureMode = posture;
   591→    this.emotionalState = emotion;
   592→    this.targetYaw = targetYaw;
   593→
   594→    const start = new THREE.Vector2(this.root.position.x, this.root.position.z);
   595→    const goal = new THREE.Vector2(targetX, targetZ);
   596→    const dist = start.distanceTo(goal);
   597→    if (dist < 0.14) {
   598→      this.waypoints = [];
   599→      return;
   600→    }
   601→
   602→    // Check if straight segment intersects central circular holo-table (r = 1.95m)
   603→    const seg = goal.clone().sub(start);
   604→    const segLenSq = seg.lengthSq();
   605→    const tProj = Math.max(0, Math.min(1, (-start.x * seg.x - start.z * seg.y) / segLenSq));
   606→    const closest = start.clone().addScaledVector(seg, tProj);
   607→    const clearanceRadius = 2.15;
   608→
   609→    this.waypoints = [];
   610→    if (closest.length() < clearanceRadius && start.length() > 1.2 && goal.length() > 1.2) {
   611→      // Compute tangent arc waypoint around the central ring
   612→      const a1 = Math.atan2(start.y, start.x);
   613→      const a2 = Math.atan2(goal.y, goal.x);
   614→      let delta = a2 - a1;
   615→      while (delta > Math.PI) delta -= Math.PI * 2;
   616→      while (delta < -Math.PI) delta += Math.PI * 2;
   617→      const midAng = a1 + delta * 0.5;
   618→      this.waypoints.push(new THREE.Vector2(Math.cos(midAng) * 2.55, Math.sin(midAng) * 2.55));
   619→    }
   620→    this.waypoints.push(goal);
   621→  }

   637→  /**
   638→   * Speak with real-time phoneme-to-Oculus-viseme facial synchronization.
   639→   * Primary voice: server-synthesized audio (base64) with the viseme timeline
   640→   * elastically scaled to the real audio duration. Fallback: browser SpeechSynthesis.
   641→   */
   642→  speak(text, { onStart, onEnd, audio } = {}) {
   643→    this.interruptSpeech();
   644→    if (!text) return;
   645→
   646→    const clean = text.replace(/\[.*?\]/g, '').trim();
   647→    if (!clean) return;
   648→
   649→    // Build phoneme/viseme schedule using LipsyncEn
   650→    const words = clean.split(/\s+/);
   651→    const timeline = [];
   652→    let cursorMs = 60;
   653→
   654→    for (const rawWord of words) {
   655→      const upper = this.lipsync.preProcessText(rawWord);
   656→      const oculusSeq = this.lipsync.wordsToVisemes(upper);
   657→      const wordDurMs = Math.max(150, Math.min(540, rawWord.length * 58));
   658→
   659→      if (oculusSeq && oculusSeq.visemes && oculusSeq.visemes.length > 0) {
   660→        const totalRel = oculusSeq.durations.reduce((a, b) => a + b, 0) || 1;
   661→        for (let i = 0; i < oculusSeq.visemes.length; i++) {
   662→          const vCode = oculusSeq.visemes[i];
   663→          const segMs = (oculusSeq.durations[i] / totalRel) * wordDurMs;
   664→          timeline.push({
   665→            viseme: `viseme_${vCode}`,
   666→            startMs: cursorMs,
   667→            endMs: cursorMs + segMs,
   668→          });
   669→          cursorMs += segMs;
   670→        }
   671→      } else {
   672→        timeline.push({ viseme: 'viseme_aa', startMs: cursorMs, endMs: cursorMs + wordDurMs * 0.7 });
   673→        cursorMs += wordDurMs;
   674→      }
   675→      cursorMs += 42;
   676→    }
   677→
   678→    this.speechTimeline = timeline;
   679→    this.speechStartPerf = performance.now();
   680→    this.isSpeaking = true;
   681→    this.onSpeechEndCallback = onEnd || null;
   682→    if (onStart) onStart();
   683→
   684→    // Primary: server voice decoded through a WebAudio AnalyserNode — the mouth
   685→    // is driven by the REAL waveform (true sync incl. pauses & emphasis), and the
   686→    // viseme clock starts the instant the buffer starts playing (zero drift).
   687→    if (audio && audio.data) {
   688→      this._playVoiceThroughAnalyser(audio).then((ok) => {
   689→        if (!ok) this._playVoiceElement(audio, clean);
   690→      });
   691→      return;
   692→    }
   693→
   694→    this._browserTtsFallback(clean);
   695→  }

   697→  _ensureAudioCtx() {
   698→    if (!this._audioCtx) {
   699→      const AC = window.AudioContext || window.webkitAudioContext;

   706→  /** Decode + play her voice through an AnalyserNode so lip-sync reads the actual
   707→   *  audio. The phoneme timeline is elastically rescaled to the real duration and
   708→   *  started together with the source. Falls back to <audio> if decode fails. */
   709→  async _playVoiceThroughAnalyser({ data, mime }) {
   710→    try {
   711→      const ctx = this._ensureAudioCtx();
   712→      if (!ctx) return false;
   713→      if (ctx.state === 'suspended') await ctx.resume();
   714→      const bin = atob(data);
   715→      const bytes = new Uint8Array(bin.length);
   716→      for (let i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
   717→      const audioBuf = await ctx.decodeAudioData(bytes.buffer);
   718→
   719→      const lastEndMs = timeline_lastEnd(this.speechTimeline);
   720→      const durMs = audioBuf.duration * 1000;
   721→      if (durMs > 400 && lastEndMs > 0) {
   722→        const factor = durMs / lastEndMs;
   723→        for (const seg of this.speechTimeline) {
   724→          seg.startMs *= factor;
   725→          seg.endMs *= factor;
   726→        }
   727→      }
   728→
   729→      const src = ctx.createBufferSource();
   730→      src.buffer = audioBuf;
   731→      const analyser = ctx.createAnalyser();
   732→      analyser.fftSize = 1024;
   733→      analyser.smoothingTimeConstant = 0.45;
   734→      src.connect(analyser);
   735→      analyser.connect(ctx.destination);
   736→      this.voiceAnalyser = analyser;
   737→      this._freqBins = new Uint8Array(analyser.frequencyBinCount);
   738→      this._timeSamples = new Float32Array(analyser.fftSize);
   739→      this.activeAudioSource = src;
   740→      this.speechStartPerf = performance.now(); // mouth clock starts WITH the sound
   741→      src.onended = () => this._onVoiceSourceEnded();
   742→      src.start();
   743→      return true;
   744→    } catch {
   745→      this.voiceAnalyser = null;
   746→      this.activeAudioSource = null;
   747→      return false;
   748→    }
   749→  }

   866→  update(dt, cameraWorldPos, reducedMotion = false) {
   867→    if (!this.loaded) return;
   868→    const clampedDt = Math.min(dt, 0.05);
   869→    this.breathClock += clampedDt;
   870→
   871→    // 1. Grounded Waypoint Locomotion (zero foot-skating!)
   872→    // In conversational focus she turns her WHOLE BODY toward you — natural
   873→    // posture for a face-to-face, and the head never tracks at its limits.
   874→    if (this.focusCamera && this.postureMode === 'conversational_focus' && this.waypoints.length === 0) {
   875→      const dx = cameraWorldPos.x - this.root.position.x;
   876→      const dz = cameraWorldPos.z - this.root.position.z;
   877→      if (Math.hypot(dx, dz) > 0.35) {
   878→        this.targetYaw = Math.atan2(dx, dz);
   879→      }
   880→    }
   881→    if (this.waypoints.length > 0) {
   882→      const wp = this.waypoints[0];
   883→      const dx = wp.x - this.root.position.x;
   884→      const dz = wp.y - this.root.position.z;
   885→      const dist = Math.hypot(dx, dz);
   886→
   887→      if (dist < 0.12) {
   888→        this.waypoints.shift();
   889→      } else {
   890→        const desiredYaw = Math.atan2(dx, dz);
   891→        this.root.rotation.y = this._lerpAngle(this.root.rotation.y, desiredYaw, clampedDt * 7.5);
   892→
   893→        const maxSpeed = this.urgentRun ? 3.25 : 1.32;
   894→        const decelFactor = this.waypoints.length === 1 ? Math.min(1, dist / 0.75) : 1;
   895→        const targetSpeed = maxSpeed * Math.max(0.32, decelFactor);
   896→        this.currentSpeed += (targetSpeed - this.currentSpeed) * Math.min(1, clampedDt * 8.0);
   897→
   898→        const step = Math.min(dist, this.currentSpeed * clampedDt);
   899→        this.root.position.x += (dx / dist) * step;
   900→        this.root.position.z += (dz / dist) * step;
   901→
   902→        const clipName = this.urgentRun && this.actions.run ? 'run' : 'walk';
   903→        this._fadeToAction(clipName, 0.22);
   904→        if (this.actions[clipName]) {
   905→          const refSpeed = clipName === 'run' ? 3.25 : 1.32;
   906→          this.actions[clipName].setEffectiveTimeScale(Math.max(0.55, this.currentSpeed / refSpeed));
   907→        }
   908→      }
   909→    } else {
   910→      this.currentSpeed *= Math.max(0, 1 - clampedDt * 10);
   911→      this._fadeToAction('idle', 0.35);
   912→      this.root.rotation.y = this._lerpAngle(this.root.rotation.y, this.targetYaw, clampedDt * 5.0);
   913→    }
   914→
   915→    // 2. Advance skeletal mixer
   916→    this.mixer.update(clampedDt);
   917→
   918→    // 2b. Camera-relative cinematic relight (key/fill/rim around her head)
   919→    this._updateCameraRelativeLighting(cameraWorldPos);
   920→
   921→    // 3. Subtle additive posture & hand interaction layering (only when standing still)
   922→    if (this.waypoints.length === 0 && !reducedMotion) {
   923→      this._applyPostureLayer();
   924→    }
   925→
   926→    // 3b. Procedural pointing at holographic evidence + living fingers
   927→    this._updateArmAim();
   928→    this._updateFingerLife(reducedMotion);
   929→
   930→    // 4. Head & Eye Gaze Tracking (rotates eyes/head naturally rather than spinning whole body!)
   931→    const gazeTarget = this.focusCamera ? cameraWorldPos : this.lookWorldTarget;
   932→    this._applyGazeTracking(gazeTarget);
   933→
   934→    // 5. Asymmetric Natural Blinking
   935→    this._updateBlinking(clampedDt);
   936→
   937→    // 6. Emotional Facial Expression Morphs
   938→    this._updateExpressions();
   939→
   940→    // 7. Synchronized Phoneme/Viseme Lip-Sync
   941→    this._updateLipSync();
   942→  }

   944→  _applyPostureLayer() {
   945→    const t = this.breathClock;
   946→    if (this.bones.Spine1) {
   947→      this.bones.Spine1.rotation.x += Math.sin(t * 1.45) * 0.012;
   948→    }
   949→
   950→    // Living weight-shift: slow hips sway with counter-rotating spine + head
   951→    // micro-drift, so the idle never reads as a frozen mannequin.
   952→    const sway = Math.sin(t * 0.31);
   953→    if (this.bones.Hips) {
   954→      this.bones.Hips.rotation.z += sway * 0.022;
   955→      this.bones.Hips.position.x += sway * 0.011;
   956→    }
   957→    if (this.bones.Spine) {
   958→      this.bones.Spine.rotation.z -= sway * 0.013;
   959→    }
   960→    if (this.bones.Head) {
   961→      this.bones.Head.rotation.y += Math.sin(t * 0.47 + 1.3) * 0.013;
   962→      this.bones.Head.rotation.z += Math.sin(t * 0.23) * 0.006;
   963→    }
   964→
   965→    const rArm = this.bones.RightArm;
   966→    const rFore = this.bones.RightForeArm;
   967→    const lArm = this.bones.LeftArm;
   968→    const lFore = this.bones.LeftForeArm;
   969→    if (!rArm || !rFore || !lArm || !lFore) return;
   970→
   971→    if (this.postureMode === 'console_interaction') {
   972→      // Right hand raised gently manipulating volumetric assembly above holo-table
   973→      rArm.rotation.x -= 0.42 + Math.sin(t * 1.1) * 0.04;
   974→      rArm.rotation.z -= 0.12;
   975→      rFore.rotation.x -= 0.55 + Math.cos(t * 1.3) * 0.05;
   976→    } else if (this.postureMode === 'scenario_construct') {
   977→      // Symmetrical restrained two-hand volumetric expansion gesture
   978→      rArm.rotation.x -= 0.48;
   979→      rFore.rotation.x -= 0.52;
   980→      lArm.rotation.x -= 0.48;
   981→      lFore.rotation.x -= 0.52;
   982→    } else if (this.postureMode === 'scenario_direct' || this.postureMode === 'video_present') {
   983→      // Measured explanatory presentation gesture toward 3D evidence
   984→      // (skipped when the aim-solver owns the arm — she POINTS at the hologram)
   985→      if (!this.gestureWorldTarget) {
   986→        rArm.rotation.x -= 0.54;
   987→        rArm.rotation.y += 0.22;
   988→        rFore.rotation.x -= 0.38;
   989→      }
   990→    } else if (this.isSpeaking) {
   991→      // Restrained conversational gesture (not on every word!)
   992→      const env = Math.max(0, Math.sin(t * 0.85));
   993→      rArm.rotation.x -= 0.22 * env;
   994→      rFore.rotation.x -= 0.38 * env;
   995→    }
   996→  }

   998→  /**
   999→   * Procedural right-arm aiming: during scenario_direct / video_present she
  1000→   * POINTS at the live holographic evidence (gestureWorldTarget, world space).
  1001→   * A quaternion slerp layered over the mixer output; ramps in/out smoothly.
  1002→   */
  1003→  _updateArmAim() {
  1004→    const arm = this.bones.RightArm;
  1005→    if (!arm || !this._rightArmRestQuat) return;
  1006→    // Cache the last valid target so the ramp-down keeps a direction after the
  1007→    // scenario clears gestureWorldTarget (prevents a null deref mid-blend).
  1008→    if (this.gestureWorldTarget) {
  1009→      (this._lastGestureTarget || (this._lastGestureTarget = new THREE.Vector3())).copy(this.gestureWorldTarget);
  1010→    }
  1011→    const target = this.gestureWorldTarget || this._lastGestureTarget;
  1012→    const aiming = (this.postureMode === 'scenario_direct' || this.postureMode === 'video_present')
  1013→      && target;
  1014→    this._armAimWeight = aiming
  1015→      ? Math.min(1, this._armAimWeight + 0.035)
  1016→      : Math.max(0, this._armAimWeight - 0.05);
  1017→    const w = this._armAimWeight;
  1018→    if (w <= 0.001 || !target) return;
  1019→
  1020→    arm.updateWorldMatrix(true, false);
  1021→    const shoulderW = arm.getWorldPosition(this._aimP || (this._aimP = new THREE.Vector3()));
  1022→    const dir = this._aimD || (this._aimD = new THREE.Vector3());
  1023→    // Gentle living sway so the pointing never reads as a frozen laser beam
  1024→    const sway = Math.sin(this.breathClock * 1.1) * 0.035;
  1025→    dir.copy(target);
  1026→    dir.y += sway;
  1027→    dir.sub(shoulderW).normalize();
  1028→
  1029→    const parentQ = this._aimQ0 || (this._aimQ0 = new THREE.Quaternion());
  1030→    arm.parent.getWorldQuaternion(parentQ);
  1031→    const invParentQ = this._aimQInv || (this._aimQInv = new THREE.Quaternion());
  1032→    invParentQ.copy(parentQ).invert();
  1033→    const localDir = this._aimL || (this._aimL = new THREE.Vector3());
  1034→    localDir.copy(dir).applyQuaternion(invParentQ).normalize();
  1035→
  1036→    const restDir = this._aimR || (this._aimR = new THREE.Vector3(0, 1, 0).applyQuaternion(this._rightArmRestQuat).normalize());
  1037→    const aimQuat = this._aimQ1 || (this._aimQ1 = new THREE.Quaternion());
  1038→    aimQuat.setFromUnitVectors(restDir, localDir).multiply(this._rightArmRestQuat);
  1039→    arm.quaternion.slerp(aimQuat, w * 0.22);
  1040→
  1041→    if (this.bones.RightForeArm) {
  1042→      this.bones.RightForeArm.rotation.x -= 0.1 * w;
  1043→    }
  1044→
  1045→    // Presenting hand: index extended, remaining fingers softly curled
  1046→    this._curlFingers('Right', ['Index'], -0.02, w);
  1047→    this._curlFingers('Right', ['Middle', 'Ring', 'Pinky'], 0.42, w);
  1048→  }

  1050→  _curlFingers(side, names, curl, w) {
  1051→    for (const n of names) {
  1052→      for (const j of [1, 2, 3]) {
  1053→        const b = this.bones[`${side}Hand${n}${j}`];
  1054→        if (b) b.rotation.x += curl * w * (j === 1 ? 0.7 : 1.0);
  1055→      }
  1056→    }
  1057→  }

  1060→  _updateFingerLife(reducedMotion) {

  1074→    }

  1077→  _applyGazeTracking(worldTarget) {
  1078→    if (!this.bones.Head || !this.bones.Neck) return;
  1079→    const headPos = this.getHeadWorldPosition(new THREE.Vector3());
  1080→    const localDir = worldTarget.clone().sub(headPos);
  1081→    localDir.applyAxisAngle(new THREE.Vector3(0, 1, 0), -this.root.rotation.y).normalize();
  1082→
  1083→    const yaw = THREE.MathUtils.clamp(Math.atan2(localDir.x, localDir.z), -0.58, 0.58);
  1084→    const pitch = THREE.MathUtils.clamp(-Math.asin(THREE.MathUtils.clamp(localDir.y, -0.8, 0.8)), -0.32, 0.32);
  1085→
  1086→    this.bones.Neck.rotation.y += yaw * 0.32;
  1087→    this.bones.Neck.rotation.x += pitch * 0.28;
  1088→    this.bones.Head.rotation.y += yaw * 0.52;
  1089→    this.bones.Head.rotation.x += pitch * 0.42;
  1090→
  1091→    // Subtle ocular look morphs
  1092→    const eyeX = THREE.MathUtils.clamp(yaw * 0.85, -0.5, 0.5);
  1093→    const eyeY = THREE.MathUtils.clamp(-pitch * 0.85, -0.4, 0.4);

  1100→    this._setMorph('eyeLookDownLeft', Math.max(0, -eyeY), 0.3);

  1104→  _updateBlinking(dt) {
  1105→    this.blinkTimer -= dt;
  1106→    if (this.blinkTimer <= 0 && this.blinkPhase < 0) {
  1107→      this.blinkPhase = 0;
  1108→      this.blinkTimer = 3.1 + Math.random() * 2.8;
  1109→    }
  1110→    if (this.blinkPhase >= 0) {
  1111→      this.blinkPhase += dt / 0.15;
  1112→      const w = this.blinkPhase < 0.5 ? this.blinkPhase * 2 : Math.max(0, (1 - this.blinkPhase) * 2);
  1113→      this._setMorph('eyeBlinkLeft', w, 0.85);
  1114→      this._setMorph('eyeBlinkRight', w * 0.96, 0.85);
  1115→      // Catchlights fade with the lids
  1116→      if (this.catchlights) {
  1117→        const cl = Math.max(0, 1 - w * 1.4);
  1118→        this.catchlights.forEach((sp, i) => {
  1119→          sp.scale.setScalar((this._catchlightBase?.[i] ?? 0.017) * cl);
  1120→        });
  1121→      }
  1122→      if (this.blinkPhase >= 1) {
  1123→        this.blinkPhase = -1;
  1124→        this._setMorph('eyeBlinkLeft', 0, 0.85);
  1125→        this._setMorph('eyeBlinkRight', 0, 0.85);
  1126→        if (this.catchlights) {
  1127→          this.catchlights.forEach((sp, i) => sp.scale.setScalar(this._catchlightBase?.[i] ?? 0.017));
  1128→        }
  1129→      }
  1130→    }
  1131→  }

  1147→  _updateLipSync() {
  1148→    if (!this.isSpeaking || this.speechTimeline.length === 0) {
  1149→      for (const v of VISEME_NAMES) this._setMorph(v, 0, 0.28);
  1150→      this._setMorph('jawOpen', 0, 0.3);
  1151→      this._setMorph('mouthOpen', 0, 0.3);
  1152→      return;
  1153→    }
  1154→
  1155→    const elapsed = performance.now() - this.speechStartPerf;
  1156→    const lastEnd = timeline_lastEnd(this.speechTimeline);
  1157→
  1158→    // --- Layer 1: phoneme timeline ("which sound") ---
  1159→    let timelineV = 'viseme_sil';
  1160→    for (const seg of this.speechTimeline) {
  1161→      if (elapsed >= seg.startMs && elapsed <= seg.endMs) {
  1162→        timelineV = seg.viseme;
  1163→        break;
  1164→      }
  1165→    }
  1166→
  1167→    // --- Layer 2: live waveform analysis ("how open, which shape") ---
  1168→    let audioJaw = null;
  1169→    let audioV = null;
  1170→    if (this.voiceAnalyser && this.activeAudioSource) {
  1171→      this.voiceAnalyser.getFloatTimeDomainData(this._timeSamples);
  1172→      this.voiceAnalyser.getByteFrequencyData(this._freqBins);
  1173→      let sum = 0;
  1174→      for (let i = 0; i < this._timeSamples.length; i++) {
  1175→        sum += this._timeSamples[i] * this._timeSamples[i];
  1176→      }
  1177→      const rms = Math.sqrt(sum / this._timeSamples.length);
  1178→      audioJaw = Math.min(0.72, rms * 3.6);
  1179→      // Crude formant estimate: spectral centroid bucket → vowel shape
  1180→      let weighted = 0;
  1181→      let total = 0;
  1182→      for (let i = 2; i < this._freqBins.length; i++) {
  1183→        const mag = this._freqBins[i] / 255;
  1184→        weighted += mag * i;
  1185→        total += mag;
  1186→      }
  1187→      if (rms < 0.018) {
  1188→        audioV = 'viseme_sil';
  1189→      } else {
  1190→        const centroid = total > 0.4 ? weighted / total : 30;
  1191→        audioV = centroid < 18 ? 'viseme_U'
  1192→          : centroid < 34 ? 'viseme_aa'
  1193→          : centroid < 70 ? 'viseme_DD'
  1194→          : 'viseme_E';
  1195→      }
  1196→    }
  1197→
  1198→    const activeV = audioV && audioV !== 'viseme_sil' ? audioV : timelineV;
  1199→    const peak = audioV && audioV !== 'viseme_sil' ? 0.95 : 0.6;
  1200→    for (const v of VISEME_NAMES) {
  1201→      this._setMorph(v, v === activeV ? peak : 0.0, 0.5);
  1202→    }
  1203→
  1204→    let jaw = timelineV === 'viseme_sil' ? 0
  1205→      : (['viseme_aa', 'viseme_O', 'viseme_E'].includes(timelineV) ? 0.3 : 0.12);
  1206→    if (audioJaw !== null) jaw = Math.max(jaw, audioJaw);
  1207→    this._setMorph('jawOpen', jaw, 0.55);
  1208→    // Extra lip aperture so articulation reads at conversational distance
  1209→    this._setMorph('mouthOpen', jaw * 0.55, 0.5);
  1210→
  1211→    // End-of-speech detection (element / browser-TTS paths; WebAudio ends via onended)
  1212→    if (this.activeAudioSource) return;
  1213→    if (elapsed > lastEnd + 260) {
  1214→      const audioStillPlaying = this.activeAudioEl && !this.activeAudioEl.paused;
  1215→      if (!audioStillPlaying && (!('speechSynthesis' in window) || !window.speechSynthesis.speaking)) {
  1216→        this.isSpeaking = false;
  1217→        this._clearVisemes();
  1218→        if (this.onSpeechEndCallback) {
  1219→          const cb = this.onSpeechEndCallback;
  1220→          this.onSpeechEndCallback = null;
  1221→          cb();
  1222→        }
  1223→      }
  1224→    }
  1225→  }

  1227→  _lerpAngle(a, b, t) {
  1228→    let diff = b - a;
  1229→    while (diff > Math.PI) diff -= Math.PI * 2;
  1230→    while (diff < -Math.PI) diff += Math.PI * 2;
  1231→    return a + diff * Math.min(1, t);
  1232→  }
