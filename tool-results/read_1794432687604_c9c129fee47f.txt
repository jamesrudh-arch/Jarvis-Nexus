    21→    21→function mulberry32(seed) {
    22→    22→  let a = seed >>> 0;
    23→    23→  return function () {
    24→    24→    a |= 0; a = (a + 0x6d2b79f5) | 0;
    25→    25→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    26→    26→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    27→    27→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    28→    28→  };
    29→    29→}

   170→   170→/**
   171→   171→ * Tiling micro-surface normal map generated from value noise (finite-difference
   172→   172→ * bump → normal). Gives skin pore-level and fabric weave specular breakup so
   173→   173→ * close-ups never read as airbrushed wax or flat cloth.
   174→   174→ */
   175→   175→export function createMicroNormalTexture({ size = 256, scale = 24, strength = 1.0, weave = false, seed = 1337 } = {}) {
   176→   176→  const height = document.createElement('canvas');
   177→   177→  height.width = height.height = size;
   178→   178→  const hCtx = height.getContext('2d');
   179→   179→  const img = hCtx.createImageData(size, size);
   180→   180→  const rand = mulberry32(seed);
   181→   181→
   182→   182→  // Multi-octave value noise field
   183→   183→  const gridN = Math.max(2, Math.round(scale));
   184→   184→  const grids = [];
   185→   185→  for (let o = 0; o < 3; o++) {
   186→   186→    const n = gridN * (1 << o);
   187→   187→    const g = new Float32Array((n + 1) * (n + 1));
   188→   188→    for (let i = 0; i < g.length; i++) g[i] = rand();
   189→   189→    grids.push({ n, g });
   190→   190→  }
   191→   191→  const sample = (x, y) => {
   192→   192→    let v = 0, amp = 1, tot = 0;
   193→   193→    for (const { n, g } of grids) {
   194→   194→      const fx = x * n, fy = y * n;
   195→   195→      const x0 = Math.floor(fx), y0 = Math.floor(fy);
   196→   196→      const tx = fx - x0, ty = fy - y0;
   197→   197→      const sx = tx * tx * (3 - 2 * tx), sy = ty * ty * (3 - 2 * ty);
   198→   198→      const i00 = g[(y0 % (n + 1)) * (n + 1) + (x0 % (n + 1))];
   199→   199→      const i10 = g[(y0 % (n + 1)) * (n + 1) + ((x0 + 1) % (n + 1))];
   200→   200→      const i01 = g[((y0 + 1) % (n + 1)) * (n + 1) + (x0 % (n + 1))];
   201→   201→      const i11 = g[((y0 + 1) % (n + 1)) * (n + 1) + ((x0 + 1) % (n + 1))];
   202→   202→      v += (i00 * (1 - sx) + i10 * sx) * (1 - sy) * amp + (i01 * (1 - sx) + i11 * sx) * sy * amp;
   203→   203→      tot += amp;
   204→   204→      amp *= 0.5;
   205→   205→    }
   206→   206→    return v / tot;
   207→   207→  };
   208→   208→
   209→   209→  for (let y = 0; y < size; y++) {
   210→   210→    for (let x = 0; x < size; x++) {
   211→   211→      const u = x / size, v = y / size;
   212→   212→      let h = sample(u, v);
   213→   213→      if (weave) {
   214→   214→        // crossed warp/weft threads
   215→   215→        const warp = Math.sin(u * Math.PI * 2 * 46) * 0.5 + 0.5;
   216→   216→        const weft = Math.sin(v * Math.PI * 2 * 46) * 0.5 + 0.5;
   217→   217→        h = h * 0.45 + Math.max(warp, weft) * 0.4 + Math.min(warp, weft) * 0.15;
   218→   218→      }
   219→   219→      img.data[(y * size + x) * 4] = img.data[(y * size + x) * 4 + 1] = img.data[(y * size + x) * 4 + 2] = h * 255;
   220→   220→      img.data[(y * size + x) * 4 + 3] = 255;
   221→   221→    }
   222→   222→  }
   223→   223→  hCtx.putImageData(img, 0, 0);
   224→   224→
   225→   225→  // Finite-difference → normal map (wrap-around for tiling)
   226→   226→  const out = document.createElement('canvas');
   227→   227→  out.width = out.height = size;
   228→   228→  const oCtx = out.getContext('2d');
   229→   229→  const oImg = oCtx.createImageData(size, size);
   230→   230→  const lum = (x, y) => img.data[(((y + size) % size) * size + ((x + size) % size)) * 4] / 255;
   231→   231→  for (let y = 0; y < size; y++) {
   232→   232→    for (let x = 0; x < size; x++) {
   233→   233→      const dx = (lum(x + 1, y) - lum(x - 1, y)) * strength;
   234→   234→      const dy = (lum(x, y + 1) - lum(x, y - 1)) * strength;
   235→   235→      const inv = 1 / Math.sqrt(dx * dx + dy * dy + 1);
   236→   236→      const i = (y * size + x) * 4;
   237→   237→      oImg.data[i] = (-dx * inv * 0.5 + 0.5) * 255;
   238→   238→      oImg.data[i + 1] = (-dy * inv * 0.5 + 0.5) * 255;
   239→   239→      oImg.data[i + 2] = inv * 255;
   240→   240→      oImg.data[i + 3] = 255;
   241→   241→    }
   242→   242→  }
   243→   243→  oCtx.putImageData(oImg, 0, 0);
   244→   244→  const tex = new THREE.CanvasTexture(out);
   245→   245→  tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
   246→   246→  applyAniso(tex);
   247→   247→  return tex;
   248→   248→}

   322→   322→/**
   323→   323→ * Remaps bright white inner-blouse pixels on Avaturn's suit texture to a tailored
   324→   324→ * midnight-slate knit (#0E1822) and deepens suit charcoal (#1C252E) to match our approved storyboard.
   325→   325→ */
   326→   326→export function tailorSuitAlbedoTexture(sourceTex) {
   327→   327→  if (!sourceTex || !sourceTex.image) return sourceTex;
   328→   328→  const img = sourceTex.image;
   329→   329→  const w = img.width || 1024;
   330→   330→  const h = img.height || 1024;
   331→   331→  const canvas = document.createElement('canvas');
   332→   332→  canvas.width = w;
   333→   333→  canvas.height = h;
   334→   334→  const ctx = canvas.getContext('2d');
   335→   335→  ctx.drawImage(img, 0, 0, w, h);
   336→   336→
   337→   337→  const imgData = ctx.getImageData(0, 0, w, h);
   338→   338→  const d = imgData.data;
   339→   339→  for (let i = 0; i < d.length; i += 4) {
   340→   340→    const r = d[i];
   341→   341→    const g = d[i + 1];
   342→   342→    const b = d[i + 2];
   343→   343→    const lum = 0.299 * r + 0.587 * g + 0.114 * b;
   344→   344→    const sat = Math.max(r, g, b) - Math.min(r, g, b);
   345→   345→
   346→   346→    if (lum > 145 && sat < 38) {
   347→   347→      // Inner white shirt -> deep midnight mock-neck knit (#0E1822) preserving fold shading
   348→   348→      const shade = lum / 255;
   349→   349→      d[i] = Math.round(14 * shade);
   350→   350→      d[i + 1] = Math.round(25 * shade);
   351→   351→      d[i + 2] = Math.round(36 * shade);
   352→   352→    } else {
   353→   353→      // Tailor grey suit -> rich architectural charcoal-slate (#1D2630)
   354→   354→      const factor = 0.62;
   355→   355→      d[i] = Math.min(255, Math.round(r * factor * 0.92));
   356→   356→      d[i + 1] = Math.min(255, Math.round(g * factor * 0.98));
   357→   357→      d[i + 2] = Math.min(255, Math.round(b * factor * 1.08));
   358→   358→    }
   359→   359→  }
   360→   360→  ctx.putImageData(imgData, 0, 0);
   361→   361→  const out = new THREE.CanvasTexture(canvas);
   362→   362→  out.flipY = sourceTex.flipY;
   363→   363→  out.colorSpace = THREE.SRGBColorSpace;
   364→   364→  applyAniso(out);
   365→   365→  out.needsUpdate = true;
   366→   366→  return out;
   367→   367→}

   404→   404→  renderFrame(t) {
   405→   405→    if (this.mode === 'explainer') return; // drawn per-state, not per-clock
   406→   406→    const { ctx, canvas } = this;
   407→   407→    const w = canvas.width;
   408→   408→    const h = canvas.height;
   409→   409→
   410→   410→    // High-contrast monochrome/ice Schlieren optical wind-tunnel field
   411→   411→    const grad = ctx.createLinearGradient(0, 0, w, h);
   412→   412→    grad.addColorStop(0, '#071017');
   413→   413→    grad.addColorStop(0.5, '#172733');
   414→   414→    grad.addColorStop(1, '#060d13');
   415→   415→    ctx.fillStyle = grad;
   416→   416→    ctx.fillRect(0, 0, w, h);
   417→   417→
   418→   418→    const cx = w * 0.48;
   419→   419→    const cy = h * 0.5;
   420→   420→    const outerR = 250;
   421→   421→    const hubR = 96;
   422→   422→
   423→   423→    ctx.strokeStyle = '#c8e4ec';
   424→   424→    ctx.lineWidth = 4;
   425→   425→    ctx.beginPath(); ctx.arc(cx, cy, outerR, 0, Math.PI * 2); ctx.stroke();
   426→   426→    ctx.beginPath(); ctx.arc(cx, cy, hubR, 0, Math.PI * 2); ctx.stroke();
   427→   427→
   428→   428→    const vortexSurge = Math.exp(-Math.pow(t - 14.2, 2) / 6.5);
   429→   429→    for (let i = 0; i < 18; i++) {
   430→   430→      const ang = (i / 18) * Math.PI * 2 + t * 0.65;
   431→   431→      const x1 = cx + Math.cos(ang) * hubR;
   432→   432→      const y1 = cy + Math.sin(ang) * hubR;
   433→   433→      const x2 = cx + Math.cos(ang + 0.14) * outerR;
   434→   434→      const y2 = cy + Math.sin(ang + 0.14) * outerR;
   435→   435→
   436→   436→      ctx.strokeStyle = 'rgba(221, 251, 255, 0.86)';
   437→   437→      ctx.lineWidth = 3;
   438→   438→      ctx.beginPath();
   439→   439→      ctx.moveTo(x1, y1);
   440→   440→      ctx.quadraticCurveTo(
   441→   441→        cx + Math.cos(ang + 0.09) * ((hubR + outerR) * 0.52),
   442→   442→        cy + Math.sin(ang + 0.09) * ((hubR + outerR) * 0.52),
   443→   443→        x2, y2
   444→   444→      );
   445→   445→      ctx.stroke();
   446→   446→
   447→   447→      ctx.strokeStyle = vortexSurge > 0.45 && i % 3 === 0
   448→   448→        ? `rgba(231, 174, 91, ${0.35 + 0.5 * vortexSurge})`
   449→   449→        : 'rgba(98, 230, 242, 0.26)';
   450→   450→      ctx.lineWidth = 1.8;
   451→   451→      ctx.beginPath();
   452→   452→      ctx.arc(cx, cy, hubR + 26 + (i % 5) * 24, ang, ang + 0.24 + vortexSurge * 0.22);
   453→   453→      ctx.stroke();
   454→   454→    }
   455→   455→
   456→   456→    if (vortexSurge > 0.35) {
   457→   457→      const vx = cx + 122;
   458→   458→      const vy = cy - 56;
   459→   459→      ctx.strokeStyle = `rgba(231, 174, 91, ${vortexSurge})`;
   460→   460→      ctx.lineWidth = 2.4;
   461→   461→      ctx.strokeRect(vx - 48, vy - 48, 96, 96);
   462→   462→      ctx.fillStyle = '#e7ae5b';
   463→   463→      ctx.font = '17px monospace';
   464→   464→      ctx.fillText('DETACHMENT LOCUS · dP/dr SPIKE', vx + 58, vy - 14);
   465→   465→    }
   466→   466→
   467→   467→    // Frame telemetry overlay
   468→   468→    ctx.fillStyle = 'rgba(2, 7, 11, 0.78)';
   469→   469→    ctx.fillRect(0, h - 52, w, 52);
   470→   470→    ctx.fillStyle = '#62e6f2';
   471→   471→    ctx.fillRect(0, h - 52, (t / this.duration) * w, 3);
   472→   472→
   473→   473→    ctx.fillStyle = '#ddfbff';
   474→   474→    ctx.font = '17px monospace';
   475→   475→    ctx.fillText(`SCHLIEREN OPTICAL BENCH · TC ${t.toFixed(2).padStart(5, '0')}s / ${this.duration.toFixed(1)}s`, 24, h - 20);
   476→   476→    ctx.fillStyle = '#7799a5';
   477→   477→    ctx.fillText(this.playing ? 'PLAYING · 60 HZ' : 'PAUSED FOR 3D EXTRACTION', w - 330, h - 20);
   478→   478→
   479→   479→    this.texture.needsUpdate = true;
   480→   480→  }

   686→   686→  _sampleSeries(s, t) {
   687→   687→    if (!s.points || !s.points.length) return null;
   688→   688→    if (t <= s.points[0].t) return s.points[0].v;
   689→   689→    const last = s.points[s.points.length - 1];
   690→   690→    if (t >= last.t) return last.v;
   691→   691→    for (let i = 1; i < s.points.length; i++) {
   692→   692→      if (s.points[i].t >= t) {
   693→   693→        const a = s.points[i - 1], b = s.points[i];
   694→   694→        const f = (t - a.t) / Math.max(0.001, b.t - a.t);
   695→   695→        return a.v + (b.v - a.v) * f;
   696→   696→      }
   697→   697→    }
   698→   698→    return last.v;
   699→   699→  }
