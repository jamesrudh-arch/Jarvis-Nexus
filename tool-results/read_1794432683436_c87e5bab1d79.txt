   335→      new THREE.CylinderGeometry(0.52, 0.4, 0.09, 32),

  1023→  renderFrame(dt, opticalVideoEl, cameraSwitchOn) {
  1024→    // 1. Procedural video & volumetric idle elements
  1025→    const scenarioActive = Boolean(this.scenarioDirector.activeScenario);
  1026→    this.explainerSidePanel.visible = scenarioActive && this.mediaScreenMat.map === this.schlierenFeed.texture;
  1027→    this.explainerSidePanelEdge.visible = this.explainerSidePanel.visible;
  1028→    if (scenarioActive && this.mediaScreenMat.map === this.schlierenFeed.texture) {
  1029→      // Documentary explainer feed synchronized to the active 3D scenario
  1030→      this.schlierenFeed.mode = 'explainer';
  1031→      this.explainerTimer += dt;
  1032→      if (this.explainerTimer > 0.1) {
  1033→        this.explainerTimer = 0;
  1034→        this.schlierenFeed.renderExplainer(this.scenarioDirector.getMediaFeedState());
  1035→      }
  1036→    } else {
  1037→      if (this.schlierenFeed.mode === 'explainer') {
  1038→        this.schlierenFeed.mode = 'schlieren';
  1039→        this.schlierenFeed.renderFrame(this.schlierenFeed.currentTime);
  1040→      }
  1041→      this.schlierenFeed.update(dt);
  1042→    }
  1043→    if (cameraSwitchOn && opticalVideoEl) {
  1044→      this.updateVisionBoard(true, opticalVideoEl);
  1045→    }
  1046→
  1047→    const holoClock = (this.holoClock = (this.holoClock || 0) + dt);
  1048→    for (const m of this.centerHoloMats) m.uniforms.uTime.value = holoClock;
  1049→    for (const m of this.idleHoloMats) m.uniforms.uTime.value = holoClock;
  1050→
  1051→    if (this.centerHoloAssembly) {
  1052→      const stageActive = Boolean(this.scenarioDirector.activeScenario);
  1053→      this.centerHoloAssembly.visible = !stageActive;
  1054→      if (!this.reducedMotion) {
  1055→        this.centerHoloAssembly.rotation.y += dt * 0.28;
  1056→        if (this.centerOrbits) {
  1057→          this.centerOrbits[0].rotation.z += dt * 0.55;
  1058→          this.centerOrbits[1].rotation.z -= dt * 0.4;
  1059→        }
  1060→      }
  1061→    }
  1062→
  1063→    if (!this.reducedMotion) {
  1064→      if (this.securityLattice) this.securityLattice.rotation.y += dt * 0.22;
  1065→      this.agentSolids.forEach((s, i) => {
  1066→        s.rotation.x += dt * (0.35 + i * 0.08);
  1067→        s.rotation.y += dt * 0.45;
  1068→      });
  1069→      if (this.extracted3DTwinGroup?.visible) {
  1070→        this.extracted3DTwinGroup.rotation.y += dt * 0.65;
  1071→      }
  1072→      // Memory constellation breathing
  1073→      for (const node of this.memoryNodes) {
  1074→        const k = 1 + 0.35 * Math.sin(holoClock * 1.6 + node.userData.pulsePhase);
  1075→        node.scale.setScalar(k);
  1076→      }
  1077→      // Dust motes drifting through the downlight cone
  1078→      if (this.dustField) {
  1079→        const pos = this.dustField.geometry.attributes.position;
  1080→        for (let i = 0; i < pos.count; i++) {
  1081→          let y = pos.getY(i) + dt * (0.045 + 0.02 * Math.sin(this.dustSeeds[i]));
  1082→          let x = pos.getX(i) + Math.sin(holoClock * 0.35 + this.dustSeeds[i]) * dt * 0.05;
  1083→          if (y > 5.4) y = 0.25;
  1084→          pos.setY(i, y);
  1085→          pos.setX(i, x);
  1086→        }
  1087→        pos.needsUpdate = true;
  1088→      }
  1089→    }
  1090→
  1091→    // 2. Update Scenario Director, Living JARVIS, Autonomous Behavior & Camera
  1092→    this.scenarioDirector.update(dt);
  1093→    this.jarvis.update(dt, this.camera.position, this.reducedMotion);
  1094→    this.behaviorDirector.update(dt, this.camera.position);
  1095→    this.cameraDirector.update(dt, this.jarvis, this.reducedMotion);
  1096→
  1097→    // 2b. Adaptive resolution guard: if the device can't hold the retina DPR,
  1098→    // step it down (never below 1.0) instead of dropping frames. Only touches
  1099→    // the pipeline when the value actually changes (no per-frame realloc churn).
  1100→    this._frameTimeAvg = this._frameTimeAvg * 0.97 + Math.min(dt, 0.2) * 1000 * 0.03;
  1101→    const maxDpr = Math.min(window.devicePixelRatio || 1, 2.0);
  1102→    let targetDpr = this._qualityDpr;
  1103→    if (this._frameTimeAvg > 42 && this._qualityDpr > 1.0) {
  1104→      targetDpr = Math.max(1.0, this._qualityDpr - 0.25);
  1105→    } else if (this._frameTimeAvg < 20 && this._qualityDpr < maxDpr) {
  1106→      targetDpr = Math.min(this._qualityDpr + 0.125, maxDpr);
  1107→    }
  1108→    if (targetDpr !== this._qualityDpr && targetDpr !== this._appliedDpr) {
  1109→      this._qualityDpr = targetDpr;
  1110→      this._appliedDpr = targetDpr;
  1111→      this.renderer.setPixelRatio(targetDpr);
  1112→      this.composer.setSize(window.innerWidth, window.innerHeight);
  1113→    }
  1114→
  1115→    // 3. Project 3D world-space scenario callout tags onto screen
  1116→    this._renderSpatialAnchors();
  1117→
  1118→    // 4. Render either standard ACES Filmic sanctuary OR 4-Face Holographic Pyramid Scissor Viewports
  1119→    if (this.hologramMode && this.hologramLayout === 'pyramid_4face') {
  1120→      this._renderFourFacePyramid();
  1121→    } else {
  1122→      this.renderer.setScissorTest(false);
  1123→      this.composer.render();
  1124→    }
  1125→  }

  1127→  _renderFourFacePyramid() {
  1128→    const w = window.innerWidth;
  1129→    const h = window.innerHeight;
  1130→    const size = Math.floor(Math.min(w, h) * 0.44);
  1131→    const cx = Math.floor(w / 2);
  1132→    const cy = Math.floor(h / 2);
  1133→
  1134→    this.renderer.setScissorTest(false);
  1135→    this.renderer.setClearColor(0x000000, 1);
  1136→    this.renderer.clear();
  1137→    this.renderer.setScissorTest(true);
  1138→
  1139→    const jPos = this.jarvis.root.position;
  1140→    const views = [
  1141→      { x: cx - size / 2, y: cy + 16, yaw: 0 },               // Top (0 deg)
  1142→      { x: cx + 16, y: cy - size / 2, yaw: Math.PI / 2 },     // Right (90 deg)
  1143→      { x: cx - size / 2, y: cy - size - 16, yaw: Math.PI },  // Bottom (180 deg)
  1144→      { x: cx - size - 16, y: cy - size / 2, yaw: -Math.PI / 2 }, // Left (270 deg)
  1145→    ];
  1146→
  1147→    for (const v of views) {
  1148→      const vx = Math.max(0, Math.floor(v.x));
  1149→      const vy = Math.max(0, Math.floor(v.y));
  1150→      const vw = Math.min(size, w - vx);
  1151→      const vh = Math.min(size, h - vy);
  1152→      if (vw <= 10 || vh <= 10) continue;
  1153→
  1154→      this.renderer.setViewport(vx, vy, vw, vh);
  1155→      this.renderer.setScissor(vx, vy, vw, vh);
  1156→      this.pyramidCam.aspect = vw / vh;
  1157→      this.pyramidCam.position.set(
  1158→        jPos.x + Math.sin(v.yaw) * 2.85,
  1159→        1.15,
  1160→        jPos.z + Math.cos(v.yaw) * 2.85
  1161→      );
  1162→      this.pyramidCam.lookAt(jPos.x, 0.98, jPos.z);
  1163→      this.pyramidCam.updateProjectionMatrix();
  1164→      this.renderer.render(this.scene, this.pyramidCam);
  1165→    }
  1166→    this.renderer.setViewport(0, 0, w, h);
  1167→    this.renderer.setScissorTest(false);

  1170→  /** Project world anchors to screen with leader dots and simple collision offsets. */
  1171→  _renderSpatialAnchors() {
  1172→    const anchors = this.scenarioDirector.worldAnchors;
  1173→    if (!anchors || anchors.length === 0 || this.hologramMode) {
  1174→      if (this.labelsContainer.childElementCount > 0) this.labelsContainer.innerHTML = '';
  1175→      return;
  1176→    }
  1177→
  1178→    while (this.labelsContainer.childElementCount < anchors.length) {
  1179→      const div = document.createElement('div');
  1180→      div.className = 'spatial-tag';
  1181→      this.labelsContainer.appendChild(div);
  1182→    }
  1183→    while (this.labelsContainer.childElementCount > anchors.length) {
  1184→      this.labelsContainer.lastElementChild.remove();
  1185→    }
  1186→
  1187→    // Project & gather
  1188→    const placed = [];
  1189→    anchors.forEach((a, i) => {
  1190→      const el = this.labelsContainer.children[i];
  1191→      const proj = a.pos.clone().project(this.camera);
  1192→      if (proj.z > 1) {
  1193→        el.style.opacity = '0';
  1194→        return;
  1195→      }
  1196→      let sx = (proj.x * 0.5 + 0.5) * window.innerWidth;
  1197→      let sy = (-proj.y * 0.5 + 0.5) * window.innerHeight;
  1198→
  1199→      // Naive collision offset: push down until free of previously placed tags
  1200→      let guard = 0;
  1201→      let clash = placed.some((p) => Math.abs(p.sx - sx) < 300 && Math.abs(p.sy - sy) < 76);
  1202→      while (clash && guard < 6) {
  1203→        sy += 66;
  1204→        guard++;
  1205→        clash = placed.some((p) => Math.abs(p.sx - sx) < 300 && Math.abs(p.sy - sy) < 76);
  1206→      }
  1207→      placed.push({ sx, sy });
  1208→
  1209→      el.className = `spatial-tag ${a.amber ? 'amber' : ''}`;
  1210→      el.style.left = `${sx}px`;
  1211→      el.style.top = `${sy}px`;
  1212→      el.style.opacity = '1';
  1213→      el.innerHTML = `${a.title}<span class="tag-sub">${a.sub}</span>`;
  1214→    });
  1215→  }
